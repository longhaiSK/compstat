### renderPlot replacement: Safari 27 (WebKit) cannot hold the call stack of
### shiny::renderPlot in webR ("Maximum call stack size exceeded"), so draw with
### renderImage instead, which nests about 45 fewer R calls
renderPlot <- function(expr, width = "auto", height = "auto", res = 72, ...,
                       env = parent.frame(), quoted = FALSE) {
    func <- shiny::exprToFunction(expr, env, quoted)
    shiny::renderImage({
        info <- shiny::getCurrentOutputInfo()
        w  <- if (is.numeric(width))  width  else info$width()
        h  <- if (is.numeric(height)) height else info$height()
        shiny::req(w > 0, h > 0)
        pr <- shiny::getDefaultReactiveDomain()$clientData$pixelratio
        if (is.null(pr)) pr <- 1
        f   <- tempfile(fileext = ".png")
        dev <- shiny:::startPNG(f, w * pr, h * pr, res = res * pr)
        on.exit(grDevices::dev.off(dev), add = TRUE)
        grDevices::dev.control(displaylist = "enable")
        v <- withVisible(func())
        if (v$visible) print(v$value)
        list(src = f, contentType = "image/png", width = w, height = h,
             coordmap = shiny:::getCoordmap(v$value, w * pr, h * pr, res * pr))
    }, deleteFile = TRUE)
}

server <- function(input, output, session) {

  ## the raw data implied by the MLE slider and n
  output$data_bb <- renderText({
    y <- round(input$mle_bb * input$n_bb)
    sprintf("Data: y = %d successes in n = %d trials (y/n = %.3f)", y, input$n_bb, y / input$n_bb)
  })
  output$prior_pg <- renderText({
    alpha <- 1 / input$disp_pg
    sprintf("Prior: Gamma(alpha = %.2f, beta = %.2f)", alpha, alpha / input$mean_pg)
  })
  output$data_pg <- renderText({
    S <- round(input$mle_pg * input$n_pg)
    sprintf("Data: sum of counts S = %d over n = %d observations (S/n = %.3f)", S, input$n_pg, S / input$n_pg)
  })
  output$data_nn <- renderText(
    sprintf("Data: sum of y_i = %.1f over n = %d observations (y-bar = %.3f)",
            input$n_nn * input$ybar_nn, input$n_nn, input$ybar_nn)
  )

  output$densplot <- renderPlot({
    par(mar = c(4, 4, 2, 4.5))
    ## likelihood (relative to its maximum) on a second y-axis, drawn on the density scale
    addlik <- function(loglik, from, to, ylim, col) {
      xs <- seq(from, to, length.out = 500)
      ll <- loglik(xs)
      L <- exp(ll - max(ll))
      k <- 0.85 * ylim[2] / 1.1          # right axis: likelihood 1 sits a little below the density peak
      lines(xs, L * k, lwd = 2, col = col, lty = 4)
      axis(4, at = c(0, 0.5, 1) * k, labels = c(0, 0.5, 1), col = col,
           col.axis = col, las = 1)
      mtext("likelihood (relative to its maximum)", side = 4, line = 3, col = col)
    }
    if (input$model == "bb") {
      a1 <- input$a1; a0 <- input$a0; n <- input$n_bb; y <- round(input$mle_bb * input$n_bb)
      post_a1 <- a1 + y; post_a0 <- a0 + (n - y)
      xs <- seq(0, 1, length.out = 500)
      dpost <- dbeta(xs, post_a1, post_a0)       # y-limit from the posterior only: the prior may be unbounded (alpha < 1)
      ylim <- c(0, max(dpost[is.finite(dpost)]) * 1.1)
      curve(dbeta(x, a1, a0), 0, 1, lwd = 2, lty = 2, col = "gray40",
            xlab = expression(theta), ylab = "density", ylim = ylim)
      curve(dbeta(x, post_a1, post_a0), add = TRUE, lwd = 3, col = "steelblue")
      addlik(function(x) dbinom(y, n, x, log = TRUE), 0, 1, ylim, "forestgreen")
      abline(v = y / n, lty = 3)
      abline(v = post_a1 / (post_a1 + post_a0), lty = 1, lwd = 1.5, col = "steelblue")
      legend("topright", bty = "n", lwd = c(2, 3, 2, 1.5), lty = c(2, 1, 4, 1),
             col = c("gray40", "steelblue", "forestgreen", "steelblue"),
             legend = c("prior", "posterior", "likelihood (right axis)", "posterior mean"))
    } else if (input$model == "pg") {
      alpha <- 1 / input$disp_pg; beta <- alpha / input$mean_pg; n <- input$n_pg; S <- round(input$mle_pg * input$n_pg)
      post_alpha <- alpha + S; post_beta <- beta + n
      hi <- 18                                  # fixed x-axis: a little beyond the largest MLE the slider allows (15)
      xs <- seq(0, hi, length.out = 500)
      dpost <- dgamma(xs, post_alpha, post_beta)  # y-limit from the posterior only: the prior may be unbounded (alpha < 1)
      ylim <- c(0, max(dpost[is.finite(dpost)]) * 1.1)
      curve(dgamma(x, alpha, beta), 0, hi, lwd = 2, lty = 2, col = "gray40",
            xlab = expression(lambda), ylab = "density", ylim = ylim)
      curve(dgamma(x, post_alpha, post_beta), add = TRUE, lwd = 3, col = "firebrick")
      addlik(function(x) dpois(S, n * x, log = TRUE), 0, hi, ylim, "forestgreen")
      abline(v = S / n, lty = 3)
      abline(v = post_alpha / post_beta, lty = 1, lwd = 1.5, col = "firebrick")
      legend("topright", bty = "n", lwd = c(2, 3, 2, 1.5), lty = c(2, 1, 4, 1),
             col = c("gray40", "firebrick", "forestgreen", "firebrick"),
             legend = c("prior", "posterior", "likelihood (right axis)", "posterior mean"))
    } else {
      mu0 <- input$mu0; kappa0 <- input$kappa0; sigma <- input$sigma_nn
      n <- input$n_nn; ybar <- input$ybar_nn
      post_kappa <- kappa0 + n
      post_mu <- (kappa0 * mu0 + n * ybar) / post_kappa
      post_sd <- sigma / sqrt(post_kappa)
      prior_sd <- sigma / sqrt(kappa0)
      lo <- -6; hi <- 6                          # fixed x-axis: a little beyond the y-bar slider range (-5 to 5)
      xs <- seq(lo, hi, length.out = 500)
      ylim <- c(0, max(dnorm(xs, mu0, prior_sd), dnorm(xs, post_mu, post_sd)) * 1.1)
      curve(dnorm(x, mu0, prior_sd), lo, hi, lwd = 2, lty = 2, col = "gray40",
            xlab = expression(mu), ylab = "density", ylim = ylim)
      curve(dnorm(x, post_mu, post_sd), add = TRUE, lwd = 3, col = "darkorchid")
      addlik(function(x) dnorm(ybar, x, sigma / sqrt(n), log = TRUE), lo, hi, ylim, "forestgreen")
      abline(v = ybar, lty = 3)
      abline(v = post_mu, lty = 1, lwd = 1.5, col = "darkorchid")
      legend("topright", bty = "n", lwd = c(2, 3, 2, 1.5), lty = c(2, 1, 4, 1),
             col = c("gray40", "darkorchid", "forestgreen", "darkorchid"),
             legend = c("prior", "posterior", "likelihood (right axis)", "posterior mean"))
    }
  })

  ## MLE vs Bayesian: estimate, standard error, 95% interval
  output$cmptab <- renderTable({
    if (input$model == "bb") {
      a1 <- input$a1; a0 <- input$a0; n <- input$n_bb; y <- round(input$mle_bb * input$n_bb)
      post_a1 <- a1 + y; post_a0 <- a0 + (n - y)
      p <- y / n; se <- sqrt(p * (1 - p) / n)
      ci <- pmin(pmax(p + c(-1, 1) * qnorm(0.975) * se, 0), 1)
      pm <- post_a1 / (post_a1 + post_a0)
      psd <- sqrt(post_a1 * post_a0 / ((post_a1 + post_a0)^2 * (post_a1 + post_a0 + 1)))
      pci <- qbeta(c(0.025, 0.975), post_a1, post_a0)
    } else if (input$model == "pg") {
      alpha <- 1 / input$disp_pg; beta <- alpha / input$mean_pg; n <- input$n_pg; S <- round(input$mle_pg * input$n_pg)
      post_alpha <- alpha + S; post_beta <- beta + n
      p <- S / n; se <- sqrt(p / n)
      ci <- pmax(p + c(-1, 1) * qnorm(0.975) * se, 0)
      pm <- post_alpha / post_beta
      psd <- sqrt(post_alpha) / post_beta
      pci <- qgamma(c(0.025, 0.975), post_alpha, post_beta)
    } else {
      mu0 <- input$mu0; kappa0 <- input$kappa0; sigma <- input$sigma_nn
      n <- input$n_nn; ybar <- input$ybar_nn
      post_kappa <- kappa0 + n
      p <- ybar; se <- sigma / sqrt(n)
      ci <- p + c(-1, 1) * qnorm(0.975) * se
      pm <- (kappa0 * mu0 + n * ybar) / post_kappa
      psd <- sigma / sqrt(post_kappa)
      pci <- qnorm(c(0.025, 0.975), pm, psd)
    }
    data.frame(
      Method = c("MLE (Wald interval)", "Bayesian (posterior)"),
      Estimate = sprintf("%.3f", c(p, pm)),
      SE = sprintf("%.3f", c(se, psd)),
      `95% interval` = c(sprintf("(%.3f, %.3f)", ci[1], ci[2]),
                         sprintf("(%.3f, %.3f)", pci[1], pci[2])),
      check.names = FALSE)
  }, striped = TRUE)

  output$summtab <- renderTable({
    if (input$model == "bb") {
      a1 <- input$a1; a0 <- input$a0; n <- input$n_bb; y <- round(input$mle_bb * input$n_bb)
      post_a1 <- a1 + y; post_a0 <- a0 + (n - y)
      logmarg    <- lbeta(post_a1, post_a0) - lbeta(a1, a0)         # exact
      pred_mean  <- post_a1 / (post_a1 + post_a0)
      pred_sd    <- sqrt(pred_mean * (1 - pred_mean))
      data.frame(
        quantity = c("prior mean", "log marginal likelihood*", "posterior distribution",
                     "predictive distribution", "predictive mean", "predictive sd"),
        value = c(sprintf("%.3f", a1 / (a1 + a0)),
                  sprintf("%.3f", logmarg),
                  sprintf("Beta(%.2f, %.2f)", post_a1, post_a0),
                  sprintf("Bernoulli(%.3f)", pred_mean),
                  sprintf("%.3f", pred_mean), sprintf("%.3f", pred_sd)))
    } else if (input$model == "pg") {
      alpha <- 1 / input$disp_pg; beta <- alpha / input$mean_pg; n <- input$n_pg; S <- round(input$mle_pg * input$n_pg)
      post_alpha <- alpha + S; post_beta <- beta + n
      logmarg   <- alpha * log(beta) - lgamma(alpha) + lgamma(post_alpha) -
        post_alpha * log(post_beta)                                 # up to -sum(log(y_i!))
      nb_prob   <- post_beta / (post_beta + 1)
      pred_mean <- post_alpha / post_beta
      pred_sd   <- sqrt(pred_mean * (1 + 1 / post_beta))
      data.frame(
        quantity = c("prior mean", "log marginal likelihood*", "posterior distribution",
                     "predictive distribution", "predictive mean", "predictive sd"),
        value = c(sprintf("%.3f", alpha / beta),
                  sprintf("%.3f", logmarg),
                  sprintf("Gamma(%.2f, %.2f)", post_alpha, post_beta),
                  sprintf("NegBin(%.2f, %.3f)", post_alpha, nb_prob),
                  sprintf("%.3f", pred_mean), sprintf("%.3f", pred_sd)))
    } else {
      mu0 <- input$mu0; kappa0 <- input$kappa0; sigma <- input$sigma_nn
      n <- input$n_nn; ybar <- input$ybar_nn
      post_kappa <- kappa0 + n
      post_mu <- (kappa0 * mu0 + n * ybar) / post_kappa
      post_sd <- sigma / sqrt(post_kappa)
      logmarg <- -n / 2 * log(2 * pi * sigma^2) + 0.5 * log(kappa0 / post_kappa) -
        kappa0 * n * (mu0 - ybar)^2 / (2 * sigma^2 * post_kappa)     # up to -SS/(2 sigma^2)
      pred_sd <- sqrt(post_sd^2 + sigma^2)
      data.frame(
        quantity = c("prior mean", "log marginal likelihood*", "posterior distribution",
                     "predictive distribution", "predictive mean", "predictive sd"),
        value = c(sprintf("%.3f", mu0),
                  sprintf("%.3f", logmarg),
                  sprintf("N(%.3f, %.3f^2)", post_mu, post_sd),
                  sprintf("N(%.3f, %.3f^2)", post_mu, pred_sd),
                  sprintf("%.3f", post_mu), sprintf("%.3f", pred_sd)))
    }
  }, colnames = FALSE, striped = TRUE)
}

shinyApp(ui, server)

About the app

Lets you explore conjugate Bayesian updating, to see how the prior, the data, and the sample size combine into the posterior. The app above lets you adjust the prior, sample size, and observed data for any of the three conjugate models and watch prior and posterior move together, along with the posterior summary table computed above.

      curve(dgamma(x, alpha, beta), 0, hi, lwd = 2, lty = 2, col = "gray40",
            xlab = expression(lambda), ylab = "density", ylim = ylim)
      curve(dgamma(x, post_alpha, post_beta), add = TRUE, lwd = 3, col = "firebrick")
      addlik(function(x) dpois(S, n * x, log = TRUE), 0, hi, ylim, "forestgreen")
      abline(v = S / n, lty = 3)
      abline(v = post_alpha / post_beta, lty = 1, lwd = 1.5, col = "firebrick")
      legend("topright", bty = "n", lwd = c(2, 3, 2, 1.5), lty = c(2, 1, 4, 1),
             col = c("gray40", "firebrick", "forestgreen", "firebrick"),
             legend = c("prior", "posterior", "likelihood (right axis)", "posterior mean"))
    } else {
      mu0 <- input$mu0; kappa0 <- input$kappa0; sigma <- input$sigma_nn
      n <- input$n_nn; ybar <- input$ybar_nn
      post_kappa <- kappa0 + n
      post_mu <- (kappa0 * mu0 + n * ybar) / post_kappa
      post_sd <- sigma / sqrt(post_kappa)
      prior_sd <- sigma / sqrt(kappa0)
      lo <- -6; hi <- 6                          # fixed x-axis: a little beyond the y-bar slider range (-5 to 5)
      xs <- seq(lo, hi, length.out = 500)
      ylim <- c(0, max(dnorm(xs, mu0, prior_sd), dnorm(xs, post_mu, post_sd)) * 1.1)
      curve(dnorm(x, mu0, prior_sd), lo, hi, lwd = 2, lty = 2, col = "gray40",
            xlab = expression(mu), ylab = "density", ylim = ylim)
      curve(dnorm(x, post_mu, post_sd), add = TRUE, lwd = 3, col = "darkorchid")
      addlik(function(x) dnorm(ybar, x, sigma / sqrt(n), log = TRUE), lo, hi, ylim, "forestgreen")
      abline(v = ybar, lty = 3)
      abline(v = post_mu, lty = 1, lwd = 1.5, col = "darkorchid")
      legend("topright", bty = "n", lwd = c(2, 3, 2, 1.5), lty = c(2, 1, 4, 1),
             col = c("gray40", "darkorchid", "forestgreen", "darkorchid"),
             legend = c("prior", "posterior", "likelihood (right axis)", "posterior mean"))
    }
  })

  ## MLE vs Bayesian: estimate, standard error, 95% interval
  output$cmptab <- renderTable({
    if (input$model == "bb") {
      a1 <- input$a1; a0 <- input$a0; n <- input$n_bb; y <- round(input$mle_bb * input$n_bb)
      post_a1 <- a1 + y; post_a0 <- a0 + (n - y)
      p <- y / n; se <- sqrt(p * (1 - p) / n)
      ci <- pmin(pmax(p + c(-1, 1) * qnorm(0.975) * se, 0), 1)
      pm <- post_a1 / (post_a1 + post_a0)
      psd <- sqrt(post_a1 * post_a0 / ((post_a1 + post_a0)^2 * (post_a1 + post_a0 + 1)))
      pci <- qbeta(c(0.025, 0.975), post_a1, post_a0)
    } else if (input$model == "pg") {
      alpha <- 1 / input$disp_pg; beta <- alpha / input$mean_pg; n <- input$n_pg; S <- round(input$mle_pg * input$n_pg)
      post_alpha <- alpha + S; post_beta <- beta + n
      p <- S / n; se <- sqrt(p / n)
      ci <- pmax(p + c(-1, 1) * qnorm(0.975) * se, 0)
      pm <- post_alpha / post_beta
      psd <- sqrt(post_alpha) / post_beta
      pci <- qgamma(c(0.025, 0.975), post_alpha, post_beta)
    } else {
      mu0 <- input$mu0; kappa0 <- input$kappa0; sigma <- input$sigma_nn
      n <- input$n_nn; ybar <- input$ybar_nn
      post_kappa <- kappa0 + n
      p <- ybar; se <- sigma / sqrt(n)
      ci <- p + c(-1, 1) * qnorm(0.975) * se
      pm <- (kappa0 * mu0 + n * ybar) / post_kappa
      psd <- sigma / sqrt(post_kappa)
      pci <- qnorm(c(0.025, 0.975), pm, psd)
    }
    data.frame(
      Method = c("MLE (Wald interval)", "Bayesian (posterior)"),
      Estimate = sprintf("%.3f", c(p, pm)),
      SE = sprintf("%.3f", c(se, psd)),
      `95% interval` = c(sprintf("(%.3f, %.3f)", ci[1], ci[2]),
                         sprintf("(%.3f, %.3f)", pci[1], pci[2])),
      check.names = FALSE)
  }, striped = TRUE)

  output$summtab <- renderTable({
    if (input$model == "bb") {
      a1 <- input$a1; a0 <- input$a0; n <- input$n_bb; y <- round(input$mle_bb * input$n_bb)
      post_a1 <- a1 + y; post_a0 <- a0 + (n - y)
      logmarg    <- lbeta(post_a1, post_a0) - lbeta(a1, a0)         # exact
      pred_mean  <- post_a1 / (post_a1 + post_a0)
      pred_sd    <- sqrt(pred_mean * (1 - pred_mean))
      data.frame(
        quantity = c("prior mean", "log marginal likelihood*", "posterior distribution",
                     "predictive distribution", "predictive mean", "predictive sd"),
        value = c(sprintf("%.3f", a1 / (a1 + a0)),
                  sprintf("%.3f", logmarg),
                  sprintf("Beta(%.2f, %.2f)", post_a1, post_a0),
                  sprintf("Bernoulli(%.3f)", pred_mean),
                  sprintf("%.3f", pred_mean), sprintf("%.3f", pred_sd)))
    } else if (input$model == "pg") {
      alpha <- 1 / input$disp_pg; beta <- alpha / input$mean_pg; n <- input$n_pg; S <- round(input$mle_pg * input$n_pg)
      post_alpha <- alpha + S; post_beta <- beta + n
      logmarg   <- alpha * log(beta) - lgamma(alpha) + lgamma(post_alpha) -
        post_alpha * log(post_beta)                                 # up to -sum(log(y_i!))
      nb_prob   <- post_beta / (post_beta + 1)
      pred_mean <- post_alpha / post_beta
      pred_sd   <- sqrt(pred_mean * (1 + 1 / post_beta))
      data.frame(
        quantity = c("prior mean", "log marginal likelihood*", "posterior distribution",
                     "predictive distribution", "predictive mean", "predictive sd"),
        value = c(sprintf("%.3f", alpha / beta),
                  sprintf("%.3f", logmarg),
                  sprintf("Gamma(%.2f, %.2f)", post_alpha, post_beta),
                  sprintf("NegBin(%.2f, %.3f)", post_alpha, nb_prob),
                  sprintf("%.3f", pred_mean), sprintf("%.3f", pred_sd)))
    } else {
      mu0 <- input$mu0; kappa0 <- input$kappa0; sigma <- input$sigma_nn
      n <- input$n_nn; ybar <- input$ybar_nn
      post_kappa <- kappa0 + n
      post_mu <- (kappa0 * mu0 + n * ybar) / post_kappa
      post_sd <- sigma / sqrt(post_kappa)
      logmarg <- -n / 2 * log(2 * pi * sigma^2) + 0.5 * log(kappa0 / post_kappa) -
        kappa0 * n * (mu0 - ybar)^2 / (2 * sigma^2 * post_kappa)     # up to -SS/(2 sigma^2)
      pred_sd <- sqrt(post_sd^2 + sigma^2)
      data.frame(
        quantity = c("prior mean", "log marginal likelihood*", "posterior distribution",
                     "predictive distribution", "predictive mean", "predictive sd"),
        value = c(sprintf("%.3f", mu0),
                  sprintf("%.3f", logmarg),
                  sprintf("N(%.3f, %.3f^2)", post_mu, post_sd),
                  sprintf("N(%.3f, %.3f^2)", post_mu, pred_sd),
                  sprintf("%.3f", post_mu), sprintf("%.3f", pred_sd)))
    }
  }, colnames = FALSE, striped = TRUE)
}
