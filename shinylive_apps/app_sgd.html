## ------------------------------------------------------------------- model
## One covariate x, four tanh hidden units, identity link, Gaussian noise:
##   a_j = w_j1 x + w_j0,  h_j = tanh(a_j),  mu(x) = v_0 + sum_j v_j h_j,
##   y | x ~ N(mu(x), sigma^2).
## theta = (w11..wJ1,  w10..wJ0,  v1..vJ,  v0)
J <- 4
NP <- 3 * J + 1                                  # number of weights
NAMES <- c(paste0("w", 1:J, "1"), paste0("w", 1:J, "0"), paste0("v", 1:J), "v0")
BIASIDX <- c(J + 1:J, NP)                        # the intercepts w_j0 and v_0
## two bumps in the mean: units 1, 2 switch on and off over (-2.2, -0.8),
## units 3, 4 over (0.8, 2.2), so mu(x) has two modes
TRUE_THETA <- c(2.5, 2.5, 2.5, 2.5,   5.5, 2, -2, -5.5,   1.2, -1.2, 1.0, -1.0,   -0.5)

fwd <- function(th, x) {
  H <- tanh(outer(x, th[1:J]) + rep(th[J + 1:J], each = length(x)))
  list(H = H, mu = as.vector(th[3 * J + 1] + H %*% th[2 * J + 1:J]))
}
mu_fun <- function(th, x) fwd(th, x)$mu

## the scaled units v_j h_j(x), one column per hidden unit
unit_terms <- function(th, x) sweep(fwd(th, x)$H, 2, th[2 * J + 1:J], "*")

## the loss: half the mean squared error (the Gaussian negative log-likelihood
## per observation, up to the constant and the factor 1 / sigma^2)
loss_fun <- function(th, x, y) {
  v <- 0.5 * mean((y - fwd(th, x)$mu)^2)
  if (is.finite(v)) v else Inf
}

## its gradient by backpropagation:  delta_mu -> delta_a -> weights
loss_grad <- function(th, x, y) {
  E <- fwd(th, x)
  d_mu <- E$mu - y                                          # n
  d_a  <- (1 - E$H^2) * outer(d_mu, th[2 * J + 1:J])        # n x J
  c(colSums(d_a * x),                                       # w_j1
    colSums(d_a),                                           # w_j0
    as.vector(crossprod(E$H, d_mu)),                        # v_j
    sum(d_mu)) / length(x)                                  # v_0
}

make_data <- function(n, sigma, seed) {
  set.seed(100 + seed)
  x <- runif(n, -3, 3)
  list(x = x, y = mu_fun(TRUE_THETA, x) + rnorm(n, 0, sigma), sigma = sigma)
}

rand_start <- function(seed) {
  set.seed(2000 + seed)
  c(rnorm(3 * J, 0, 1), 0)
}

## ------------------------------------------------------- Wolfe line search
line_search <- function(f, g, x, d, a0 = 1, c1 = 1e-4, c2 = 0.9) {
  f0 <- f(x); gd0 <- sum(g(x) * d)
  if (!is.finite(gd0) || gd0 >= 0) return(list(alpha = 0, ok = FALSE))
  lo <- 0; hi <- Inf; a <- a0
  for (it in 1:60) {
    fa <- f(x + a * d)
    if (!is.finite(fa) || fa > f0 + c1 * a * gd0) {
      hi <- a; a <- 0.5 * (lo + hi)
    } else {
      if (sum(g(x + a * d) * d) < c2 * gd0) {
        lo <- a
        a <- if (is.finite(hi)) 0.5 * (lo + hi) else min(2 * a, 1e4)
      } else {
        return(list(alpha = a, ok = TRUE))
      }
    }
    if (a < 1e-14) return(list(alpha = 0, ok = FALSE))
  }
  list(alpha = a, ok = (a > 0))
}

## ------------------------------------------------------------- algorithms
## the methods that step through mini-batches: one iteration is one epoch
MINIBATCH <- c("sgdls", "mom", "adam")
## Row k of P, G, F is the state after k - 1 steps; A[k] is the step length
## that led to it.
run_alg <- function(alg, th0, x, y, maxit, lr = 0.1, mom = 0.9, m = 20,
                    tol = 1e-6) {
  p <- length(th0)
  f <- function(th) loss_fun(th, x, y)
  g <- function(th) loss_grad(th, x, y)
  P <- G <- matrix(NA_real_, maxit + 1, p)
  Fv <- AL <- rep(NA_real_, maxit + 1)
  th <- th0; B <- diag(p)
  g_prev <- d_prev <- s_prev <- NULL
  n <- 0; note <- "maximum number of iterations reached"
  ## state for the fixed-rate rules
  n_obs <- length(x); m <- max(1, min(m, n_obs))
  capped <- FALSE
  if (alg %in% MINIBATCH) {                  # keep the number of mini-batch steps in check
    cap <- max(1, floor(20000 / ceiling(n_obs / m)))
    if (maxit > cap) { maxit <- cap; capped <- TRUE }
  }
  vel <- m1 <- m2 <- numeric(p)              # momentum velocity; Adam moments
  tt <- 0                                    # mini-batch updates so far (Adam)
  if (alg %in% MINIBATCH) set.seed(11)

  for (k in 1:(maxit + 1)) {
    gk <- g(th); fk <- f(th)
    P[k, ] <- th; G[k, ] <- gk; Fv[k] <- fk; n <- k
    if (!is.finite(fk) || !all(is.finite(gk))) { note <- "diverged"; break }
    if (sqrt(sum(gk^2)) < tol) { note <- "converged: gradient norm < 1e-6"; break }
    if (k > maxit) break

    if (alg == "gdfix") {                    # full-batch gradient, fixed learning rate
      AL[k + 1] <- lr
      th <- th - lr * gk
      next
    }

    if (alg %in% MINIBATCH) {                # one iteration is one epoch: a shuffled pass
      perm <- sample.int(n_obs)              # of mini-batch steps over all the data, so
      al <- numeric(0)                       # it costs about one full-batch gradient
      for (s0 in seq(1, n_obs, by = m)) {
        idx <- perm[s0:min(s0 + m - 1, n_obs)]
        xb <- x[idx]; yb <- y[idx]
        gb <- loss_grad(th, xb, yb)               # the gradient of mini-batch B_k
        if (alg == "sgdls") {                # backtracking line search on the batch loss
          fb <- loss_fun(th, xb, yb); a <- lr; gg <- sum(gb^2)
          for (j in 1:30) {
            fa <- loss_fun(th - a * gb, xb, yb)
            if (is.finite(fa) && fa <= fb - 1e-4 * a * gg) break
            a <- a / 2
          }
          al <- c(al, a)
          th <- th - a * gb
        } else if (alg == "mom") {           # heavy ball on the batch gradient
          vel <- mom * vel - lr * gb
          th <- th + vel
        } else {                             # Adam on the batch gradient
          tt <- tt + 1
          m1 <- 0.9 * m1 + 0.1 * gb
          m2 <- 0.999 * m2 + 0.001 * gb^2
          th <- th - lr * (m1 / (1 - 0.9^tt)) / (sqrt(m2 / (1 - 0.999^tt)) + 1e-8)
        }
      }
      AL[k + 1] <- if (alg == "sgdls") mean(al) else lr
      next
    }

    if (alg == "gd") {
      d <- -gk
    } else if (alg == "cg") {
      if (is.null(d_prev)) {
        d <- -gk
      } else {
        yk <- gk - g_prev
        beta <- max(0, sum(gk * yk) / max(sum(g_prev^2), 1e-300))   # PR+
        d <- -gk + beta * d_prev
      }
      if (sum(gk * d) > -1e-12) d <- -gk                           # restart
    } else {                                                       # BFGS
      if (!is.null(s_prev)) {
        yk <- gk - g_prev; sy <- sum(s_prev * yk)
        if (sy > 1e-10) {
          if (k == 2) B <- (sy / sum(yk^2)) * diag(p)    # scale the first B
          rho <- 1 / sy
          V <- diag(p) - rho * outer(s_prev, yk)
          B <- V %*% B %*% t(V) + rho * outer(s_prev, s_prev)
        }
      }
      d <- -as.vector(B %*% gk)
      if (sum(gk * d) > -1e-12) { B <- diag(p); d <- -gk }
    }

    a0 <- if (alg == "bfgs" && k > 1) 1 else min(1, 1 / sqrt(sum(d^2)))
    ls <- line_search(f, g, th, d, a0)
    if (!ls$ok) { note <- "line search failed"; break }
    AL[k + 1] <- ls$alpha
    s_prev <- ls$alpha * d; g_prev <- gk; d_prev <- d
    th <- th + s_prev
  }
  if (capped && grepl("^maximum", note))
    note <- paste0("stopped after ", maxit, " epochs (20000 mini-batch steps)")
  i <- seq_len(n)
  list(P = P[i, , drop = FALSE], G = G[i, , drop = FALSE], F = Fv[i],
       A = AL[i], n = n, note = note)
}

ALGS <- c("Gradient descent, fixed learning rate"     = "gdfix",
          "Gradient descent + line search"            = "gd",
          "Conjugate gradient (Polak-Ribiere+)"       = "cg",
          "BFGS (quasi-Newton)"                       = "bfgs",
          "SGD + line search (on each mini-batch)"    = "sgdls",
          "SGD + momentum"                            = "mom",
          "SGD + Adam"                                = "adam")

alg_name <- function(a) if (a == "none") "none" else names(ALGS)[ALGS == a]

## --------------------------------------------------------- network layout
## Weights sit on the edges; each intercept is a node of its own, a box with
## the value and its arrows, hung on the neuron it shifts.
NX <- c(0.07, 0.50)                                 # (x, y) of the input node
YH <- seq(0.88, 0.12, length.out = J); XH <- 0.60   # hidden nodes
NO <- c(1.13, 0.50)                                 # output node
at <- function(p, q, f) p + f * (q - p)             # a point along an edge
## hidden intercepts hang above the upper half of the hidden nodes and below the rest
YB <- YH + rep(c(0.14, -0.14), each = J / 2)
BIAS <- c(rep(0.56, J), 1.00)                       # x of the intercept boxes
BIASY <- c(YB, 0.17)                                # y of the intercept boxes
## where each weight's value and arrows sit
ADJ <- local({
  hj <- lapply(1:J, function(j) c(XH, YH[j]))
  pos <- c(lapply(1:J, function(j) at(NX, hj[[j]], 0.50)),    # w_j1: x -> h_j
           lapply(1:J, function(j) c(BIAS[j], BIASY[j])),     # w_j0: box on h_j
           lapply(1:J, function(j) at(hj[[j]], NO, 0.50)),    # v_j: h_j -> out
           list(c(BIAS[J + 1], BIASY[J + 1])))                # v_0: box on output
  data.frame(idx = 1:NP, sym = NAMES,
             x = sapply(pos, `[`, 1), y = sapply(pos, `[`, 2),
             stringsAsFactors = FALSE)
})
## the arrows beside each value (also the click targets): up / down beside an
## edge weight, left / right beside an intercept box
AX <- 0.104                                  # arrow offset from the value
isb <- ADJ$idx %in% BIASIDX
ARROWS <- rbind(
  data.frame(idx = ADJ$idx, dir = +1, horiz = isb,
             ax = ifelse(isb, ADJ$x + AX, ADJ$x + AX),
             ay = ifelse(isb, ADJ$y, ADJ$y + 0.028)),
  data.frame(idx = ADJ$idx, dir = -1, horiz = isb,
             ax = ifelse(isb, ADJ$x - AX, ADJ$x + AX),
             ay = ifelse(isb, ADJ$y, ADJ$y - 0.028)))

## compander: maps v in [-vmax, vmax] to [-1, 1], boosting small values
tone <- function(v, vmax, k) {
  u <- pmax(-1, pmin(1, v / vmax))
  sign(u) * log1p(k * abs(u)) / log1p(k)
}
hcol <- function(v, vmax, k) DIV[round((tone(v, vmax, k) + 1) * 100) + 1]

draw_net <- function(v, vmax, k, main, arrows = TRUE, digits = 2) {
  col <- hcol(v, vmax, k); lw <- 0.8 + 6 * abs(tone(v, vmax, k))
  plot.new(); plot.window(c(0, 1.2), c(-0.25, 1.08))
  title(main = main, cex.main = 1.05)
  H <- cbind(XH, YH)
  ## edges, each carrying one weight: x -> h_j (w_j1), h_j -> output (v_j)
  for (j in 1:J) {
    segments(NX[1], NX[2], H[j, 1], H[j, 2], col = col[j],     lwd = lw[j],     lend = 1)
    segments(H[j, 1], H[j, 2], NO[1], NO[2], col = col[2 * J + j], lwd = lw[2 * J + j], lend = 1)
  }
  ## intercept nodes: a thin arrow from each box to the neuron it shifts
  tgt <- rbind(H, NO)
  for (i in seq_along(BIASIDX)) {
    b <- ADJ[BIASIDX[i], ]
    dy <- sign(tgt[i, 2] - b$y)
    if (i == J + 1) arrows(b$x, b$y + 0.03, NO[1] - 0.03, NO[2] - 0.03, length = 0.08,
                       col = "grey45", lwd = 1.4)
    else arrows(b$x, b$y + dy * 0.035, tgt[i, 1], tgt[i, 2] - dy * 0.035,
                length = 0.08, col = "grey45", lwd = 1.4)
  }
  ## nodes
  points(rbind(NX, H, NO), pch = 21, cex = 3.2, bg = "white", lwd = 1.6)
  text(NX[1], NX[2], "x", font = 2)
  text(H[, 1], H[, 2], paste0("h", 1:J), cex = 0.8)
  text(NO[1], NO[2], expression(mu), cex = 1.1)
  ## values: an intercept is a box coloured by its value; an edge weight is
  ## plain text on a white patch so the edge does not strike through it
  for (i in 1:NP) {
    b <- ADJ[i, ]; isb <- i %in% BIASIDX
    if (isb) rect(b$x - 0.075, b$y - 0.036, b$x + 0.075, b$y + 0.036,
                  col = col[i], border = "grey35")
    else rect(b$x - 0.072, b$y - 0.028, b$x + 0.072, b$y + 0.028,
              col = adjustcolor("white", 0.8), border = NA)
    dark <- abs(tone(v[i], vmax, k)) > 0.62
    text(b$x, b$y, sprintf(paste0("%s=%.", digits, "f"), b$sym, v[i]),
         cex = 0.85, font = if (isb) 1 else 2,
         col = if (isb) { if (dark) "white" else "black" }
               else if (dark) col[i] else "grey20")
  }
  if (arrows) for (r in seq_len(nrow(ARROWS))) {
    a <- ARROWS[r, ]
    if (a$horiz)                                   # left / right: -, +
      polygon(a$ax + a$dir * c(-0.013, -0.013, 0.015),
              a$ay + c(-0.033, 0.033, 0), col = "grey30", border = NA)
    else                                           # down / up: -, +
      polygon(a$ax + c(-0.026, 0.026, 0), a$ay + a$dir * c(-0.016, -0.016, 0.02),
              col = "grey30", border = NA)
  }
  ## signed colour bar
  xs <- seq(0.12, 1.08, length.out = 201)
  rect(xs, -0.24, xs + diff(xs)[1], -0.19, col = DIV, border = NA)
  lab <- formatC(vmax, format = "g", digits = 2)
  text(c(0.12, 0.60, 1.08), -0.16, c(paste0("-", lab), "0", paste0("+", lab)),
       cex = 0.72)
}

server <- function(input, output, session) {

  theta0 <- reactiveVal(rand_start(1))        # the starting weights

  observeEvent(input$newdata, dseed(dseed() + 1))
  ## a sensible starting learning rate for each fixed-rate method
  for (s in c("A", "B")) local({
    s <- s
    observeEvent(input[[paste0("alg", s)]], {
      lr0 <- c(gdfix = 0.3, sgdls = 0.3, mom = 0.03, adam = 0.1)[input[[paste0("alg", s)]]]
      if (!is.na(lr0)) updateSliderInput(session, paste0("lr", s), value = unname(lr0))
    })
  })
  observeEvent(input$setzero, theta0(rep(0, NP)))
  observeEvent(input$newstart, {
    rseed(rseed() + 1); theta0(rand_start(rseed()))
  })
  ## an arrow click moves the nearest arrow's weight by one step
  observeEvent(input$netv_click, {
    cl <- input$netv_click
    d <- ((cl$x - ARROWS$ax) / 0.05)^2 + ((cl$y - ARROWS$ay) / 0.04)^2
    i <- which.min(d)
    if (d[i] < 2) {
      th <- theta0()
      th[ARROWS$idx[i]] <- th[ARROWS$idx[i]] + ARROWS$dir[i] * as.numeric(input$astep)
      theta0(th)
    }
  })

  ## the predictor on a different scale: x -> s x, with the slope weights of the generating
  ## network divided by s, so the true curve is the same function of x / s
  xsc <- reactive(if (is.null(input$xscale)) 1 else as.numeric(input$xscale))
  TT <- reactive({ th <- TRUE_THETA; th[1:J] <- th[1:J] / xsc(); th })
  observeEvent(input$settrue, theta0(TT()))
  DAT <- reactive({
    D <- make_data(input$n, as.numeric(input$sigma), dseed())
    D$x <- D$x * xsc()
    D
  })
  ## one run for each optimizer, from the same start on the same data

    D <- DAT(); a <- input[[paste0("alg", s)]]
    if (is.null(a) || a == "none") return(NULL)
    fixed <- a %in% c("gdfix", "sgdls", "mom", "adam")

            lr  = if (fixed) input[[paste0("lr", s)]] else NA,
            mom = if (a == "mom") input[[paste0("mom", s)]] else NA,

  })
  runA <- mkrun("A"); runB <- mkrun("B")
  ## the network panels show the run chosen in "Network of"
  netsel <- reactive(if (identical(input$netsel, "B") && !is.null(runB())) "B" else "A")
  run <- reactive(if (netsel() == "B") runB() else runA())

  ## the browser owns the iteration (slider, play timer, buttons); a new run restarts it
  observeEvent(list(runA(), runB()), {
    nmax <- max(runA()$n, if (is.null(runB())) 0L else runB()$n)
    session$sendCustomMessage("krun", list(max = nmax - 1L,
                                           play = isTRUE(input$autoplay) && nmax > 1))
  })
  kraw <- reactive(as.integer(if (is.null(input$kshow)) 0L else input$kshow))
  kpos <- reactive(min(kraw(), run()$n - 1L))

  cur <- reactive({
    R <- run(); k <- min(kpos(), R$n - 1L) + 1L
    list(R = R, k = k, th = R$P[k, ], g = R$G[k, ])
  })

  output$netv <- renderPlot({
    C <- cur(); R <- C$R
    vmax <- max(3, quantile(abs(R$P), 0.95))
    par(mar = c(1, 1, 3, 1))
    draw_net(C$th, vmax, as.numeric(input$ctr),
             sprintf("%s: weights, iteration %d   (loss %.4f)", netsel(), C$k - 1L, R$F[C$k]))
  })

  output$netg <- renderPlot({
    C <- cur()
    gmax <- max(1e-8, max(abs(C$g)))
    par(mar = c(1, 1, 3, 1))
    draw_net(C$g, gmax, as.numeric(input$ctr),
             sprintf("%s: gradient of the loss, iteration %d", netsel(), C$k - 1L),
             arrows = FALSE, digits = 3)
  })

  ## one clearly different colour per hidden unit
  PCOL <- c("#F28E00", "#1F5FBF", "black", "#00B5CC")

  output$fit <- renderPlot({
    RA <- runA(); RB <- runB(); R <- RA; D <- DAT(); kk <- kraw()
    xg <- seq(-3, 3, length.out = 300) * xsc()
    par(mar = c(4, 4.2, 2.5, 1))
    plot(D$x, D$y, pch = 19, cex = 0.7, col = adjustcolor("#E8820C", 0.5),
         ylim = range(D$y, mu_fun(TT(), xg)),
         xlab = "x", ylab = expression(mu(x) ~ "and data y"),
         main = "fitted mean and the generated data")
    lines(xg, mu_fun(TT(), xg), col = "red", lty = 2, lwd = 2)
    lines(xg, mu_fun(R$P[1, ], xg), col = "grey60", lwd = 2)
    lines(xg, mu_fun(RA$P[min(kk, RA$n - 1L) + 1L, ], xg), col = "#2E8B57", lwd = 3)
    if (!is.null(RB))
      lines(xg, mu_fun(RB$P[min(kk, RB$n - 1L) + 1L, ], xg), col = "#1F5FBF", lwd = 2.5)
    leg <- data.frame(
      lab = c("true mean", "starting weights", paste("A:", alg_name(input$algA)), "data"),
      lwd = c(2, 2, 3, NA), lty = c(2, 1, 1, NA), pch = c(NA, NA, NA, 19),
      col = c("red", "grey60", "#2E8B57", adjustcolor("#E8820C", 0.7)),

  })

    legend("topright", bty = "n", cex = 0.78, ncol = 2,
           lwd = c(rep(2.5, J), 3, 2), lty = c(rep(3, J), 1, 2),

  output$conv <- renderPlot({

About the app

Fits a Gaussian neural network (a regression with normal noise and an identity link), with one covariate and four tanh hidden units, to data generated from known weights plus normal noise whose standard deviation you choose, and lets you watch seven optimizers work on it: gradient descent with a fixed learning rate, gradient descent with a line search, conjugate gradient and BFGS, which use the full data, and SGD with a line search, SGD with momentum and SGD with Adam, which step through mini-batches. The four units are enough for a mean \(\mu(x)\) with two modes: the true weights switch units 1 and 2 on and off around \(x \in (-2.2, -0.8)\), and units 3 and 4 around \((0.8, 2.2)\).

The weights. The network panels belong to one optimizer, chosen by Network of above them. The upper network shows the thirteen weights at the current iteration. The eight weights on edges (w11, …, w41 from \(x\) to the hidden units and v1, …, v4 from the hidden units to the output) are written on their edges, and each intercept (w10, …, w40 and v0) is a node of its own, a box hung on the neuron it shifts. Colour shows the value (blue negative, red positive; on an edge it also sets the width). The arrows beside a value change that weight (up and down on an edge, right to increase and left to decrease on an intercept box), and the starting point of the run changes with it. True Value loads the weights that generated the data, Random draws a new set, and Zero sets every weight to 0 (a saddle point of the loss: with all hidden weights at zero, only \(v_0\) has a nonzero gradient). Adjust Size sets how much one click on an arrow changes a weight.

The methods. Write \(g_k = \nabla L(\theta_k)\) for the gradient of the average loss \(L\) over all \(n\) observations at the current weights, \(\ell_i\) for the loss of observation \(i\), and \(g_k^{(B_k)} = \frac{1}{m}\sum_{i \in B_k} \nabla \ell_i(\theta_k)\) for the gradient of the mini-batch \(B_k\) of \(m\) observations used at update \(k\) (the superscript names the batch, the subscript the iteration). The first four optimizers in the menu use the full data; the last three step through mini-batches. All of them can be written as one update

\[\theta_{k+1} = \theta_k - \alpha_k\, P_k\, m_k ,\]

where \(m_k\) is the gradient actually used (the full gradient, a mini-batch gradient, or an average of past ones), \(P_k\) is a preconditioner that rescales or rotates the step, and \(\alpha_k\) is the step size.

Optimizer Gradient used, \(m_k\) Preconditioner, \(P_k\) Step size, \(\alpha_k\)
GD, fixed rate \(g_k\) \(I\) \(\alpha\), set by hand
GD + line search \(g_k\) \(I\) \(\alpha_k\) from a Wolfe line search
Conjugate gradient (PR+) \(d_k = -g_k + \beta_k d_{k-1}\), with \(\beta_k = \max\!\Big(0, \dfrac{g_k^\top (g_k - g_{k-1})}{g_{k-1}^\top g_{k-1}}\Big)\) implicit: the new gradient is combined with the previous direction \(\alpha_k\) from a Wolfe line search
BFGS \(g_k\) \(B_k \approx H^{-1}\), a full matrix, updated by \(B_{k+1} = (I - \rho\, s y^\top) B_k (I - \rho\, y s^\top) + \rho\, s s^\top\) with \(s = \theta_{k+1} - \theta_k\), \(y = g_{k+1} - g_k\), \(\rho = 1/(s^\top y)\) \(\alpha_k\) from a Wolfe line search
SGD + line search \(g_k^{(B_k)}\), with \(E\,g_k^{(B_k)} = g_k\) \(I\) \(\alpha_k\) by backtracking on the loss of the mini-batch \(B_k\): start at the learning rate and halve until \(L^{(B_k)}(\theta_k - \alpha g_k^{(B_k)}) \le L^{(B_k)}(\theta_k) - c_1 \alpha \|g_k^{(B_k)}\|^2\)
SGD + momentum \(v_k = \mu\, v_{k-1} + g_k^{(B_k)}\), an exponentially weighted sum of past mini-batch gradients \(I\) \(\alpha\), set by hand
SGD + Adam \(\hat m_k = \dfrac{(1-\beta_1)\sum_{j \le k} \beta_1^{k-j} g_j^{(B_j)}}{1-\beta_1^{k}}\), a bias-corrected average of past mini-batch gradients \(\mathrm{diag}\big(1/(\sqrt{\hat v_k} + \epsilon)\big)\), with \(\hat v_k = \dfrac{(1-\beta_2)\sum_{j \le k} \beta_2^{k-j} g_j^{(B_j)} \odot g_j^{(B_j)}}{1-\beta_2^{k}}\) \(\alpha\), set by hand

In the app \(\beta_1 = 0.9\), \(\beta_2 = 0.999\) and \(\epsilon = 10^{-8}\) for Adam, and \(k\) counts mini-batch updates. Momentum is coded as \(v \leftarrow \mu v - \alpha g^{(B)}\), \(\theta \leftarrow \theta + v\), the same update written with the sign and \(\alpha\) absorbed into \(v\). A Wolfe line search takes the step \(\alpha\) along \(d_k = -P_k m_k\) that satisfies both \(L(\theta_k + \alpha d_k) \le L(\theta_k) + c_1 \alpha\, g_k^\top d_k\) (sufficient decrease) and \(\nabla L(\theta_k + \alpha d_k)^\top d_k \ge c_2\, g_k^\top d_k\) (curvature), with \(c_1 = 10^{-4}\) and \(c_2 = 0.9\).

What each one changes. Fixed-rate GD and the line-search methods use the exact gradient. A line search removes the learning rate to tune; conjugate gradient and BFGS also change the direction, CG by combining the gradient with the previous direction and BFGS by learning an approximation of the inverse Hessian from gradient differences, the only full-matrix preconditioner here (compare Newton’s method, \(P_k = H^{-1}\)). The three SGD methods replace the gradient by a cheap and noisy mini-batch gradient. SGD with a line search chooses each step by backtracking on the loss of the batch itself, so the step is adapted to the batch and not to the whole loss. Momentum averages the mini-batch gradients, which damps their noise and the zig-zag across a valley. Adam also divides each coordinate by the square root of a running average of its squared gradients, a per-coordinate step size; why this helps so much is not fully understood (see the book’s discussion).

Epochs and mini-batches. For the three SGD methods one iteration is one epoch: a shuffled pass of \(\lceil n/m \rceil\) mini-batch updates over all the data. An iteration therefore costs about as many gradient evaluations as one full-batch step of the other methods, and the iteration axes are directly comparable. This is the point of the comparison. After the same number of passes through the data, the mini-batch methods have made \(n/m\) updates per pass where gradient descent has made one, so on a large sample they are well ahead of a full-batch method such as GD + line search in the plots. With the batch size equal to \(n\) the three methods use the full gradient. The loss and gradient norm shown are always those of the full data. The sample size goes up to 2000: the larger \(n\) is, the more mini-batch updates an epoch holds (\(n/m\)), and the clearer the advantage over the full-batch methods becomes. For these thirteen parameters and a small sample, BFGS is hard to beat; SGD, momentum and Adam are what larger networks use, where a Hessian approximation is too expensive and a pass through all the data is rarely made. To keep the browser responsive a mini-batch run stops after 20000 updates in all, which the info line reports.

Scale of \(x\). The predictor is multiplied by this factor and the slope weights of the generating network are divided by it, so the true curve is the same function of \(x\) divided by the scale and only the parameterization changes. The gradient with respect to a slope weight \(w_{j1}\) carries a factor \(x\) and the gradient with respect to an intercept does not, so a scale far from 1 makes the gradients of the weights differ by orders of magnitude. A single learning rate is then right for some weights and wrong for others: fixed-rate GD and momentum must shrink it to the most sensitive weight, while Adam, which rescales each weight separately, is much less affected. Try 10 or 100 with momentum against Adam, and with True Value as the start. The line-search methods are less sensitive, because the step length adapts, but BFGS and CG slow down when the problem is badly scaled. Changing the scale restarts the run.

The gradient. The lower network has the same layout, with each edge and intercept box coloured by the corresponding component of the gradient of the loss, computed by backpropagation. Its colour scale is set by the largest component at the current iteration, so you can see which weights the method is still pushing on as the run converges.

The fit. The right panel shows the generated data, the true mean curve (dashed red, with its two modes), the mean curve at the starting weights (grey) and the current fit (green). Two optimizers, A (green) and B (blue, or None), are run from the same start on the same data, each with its own settings, and both fits are drawn at the same iteration, so any two of the methods can be compared; the loss and gradient-norm plots show both paths. The default pair is SGD with Adam against gradient descent with a line search, which shows what the mini-batch updates buy per pass through the data. With an identity link the units add directly on the scale of the data, \(\mu(x) = v_0 + \sum_j v_j h_j(x)\), so the second plot shows the four scaled units \(v_j h_j\) (dotted; orange, blue, black and cyan for units 1 to 4) and the fit they add up to together with \(v_0\) (green) against the true mean (dashed red). The box above the plots hides this second plot. Beneath the fit are the loss (half the mean squared error) and the gradient norm per iteration; the loss plot is on a log10 scale by default, since the loss is often close to its floor and the methods differ only in the last digits (the buttons under it switch to the loss itself, or to the log of the loss minus the smallest value either run reaches, which makes the rate of convergence readable), and its dashed horizontal line is the noise floor \(\sigma^2/2\), the smallest average loss a correct model can reach, so you can tell a good fit from one that is merely still improving. The first row of controls sets up the data (sample size, noise level and a new dataset); the next two rows choose optimizers A and B and the settings of each (learning rate, momentum, mini-batch size); the last row holds the maximum number of iterations, the colour contrast, the playback buttons, and a slider that shows the fit at any iteration of the run (it follows the playback, and you can drag it to scrub through the run). The loss is not convex, so different starts can end at different fits; try Random a few times.

A Poisson version of this app, for count data with a log link, is Fitting a Poisson Neural Network by Gradient Methods.

## ------------------------------------------------------------------- model
## One covariate x, four tanh hidden units, identity link, Gaussian noise:
##   a_j = w_j1 x + w_j0,  h_j = tanh(a_j),  mu(x) = v_0 + sum_j v_j h_j,
##   y | x ~ N(mu(x), sigma^2).
## theta = (w11..wJ1,  w10..wJ0,  v1..vJ,  v0)
J <- 4
NP <- 3 * J + 1                                  # number of weights
NAMES <- c(paste0("w", 1:J, "1"), paste0("w", 1:J, "0"), paste0("v", 1:J), "v0")
BIASIDX <- c(J + 1:J, NP)                        # the intercepts w_j0 and v_0
## two bumps in the mean: units 1, 2 switch on and off over (-2.2, -0.8),
## units 3, 4 over (0.8, 2.2), so mu(x) has two modes
TRUE_THETA <- c(2.5, 2.5, 2.5, 2.5,   5.5, 2, -2, -5.5,   1.2, -1.2, 1.0, -1.0,   -0.5)

fwd <- function(th, x) {
  H <- tanh(outer(x, th[1:J]) + rep(th[J + 1:J], each = length(x)))
  list(H = H, mu = as.vector(th[3 * J + 1] + H %*% th[2 * J + 1:J]))
}
mu_fun <- function(th, x) fwd(th, x)$mu

## the scaled units v_j h_j(x), one column per hidden unit
unit_terms <- function(th, x) sweep(fwd(th, x)$H, 2, th[2 * J + 1:J], "*")

## the loss: half the mean squared error (the Gaussian negative log-likelihood
## per observation, up to the constant and the factor 1 / sigma^2)
loss_fun <- function(th, x, y) {
  v <- 0.5 * mean((y - fwd(th, x)$mu)^2)
  if (is.finite(v)) v else Inf
}

## its gradient by backpropagation:  delta_mu -> delta_a -> weights
loss_grad <- function(th, x, y) {
  E <- fwd(th, x)
  d_mu <- E$mu - y                                          # n
  d_a  <- (1 - E$H^2) * outer(d_mu, th[2 * J + 1:J])        # n x J
  c(colSums(d_a * x),                                       # w_j1
    colSums(d_a),                                           # w_j0
    as.vector(crossprod(E$H, d_mu)),                        # v_j
    sum(d_mu)) / length(x)                                  # v_0
}

make_data <- function(n, sigma, seed) {
  set.seed(100 + seed)
  x <- runif(n, -3, 3)
  list(x = x, y = mu_fun(TRUE_THETA, x) + rnorm(n, 0, sigma), sigma = sigma)
}

rand_start <- function(seed) {
  set.seed(2000 + seed)
  c(rnorm(3 * J, 0, 1), 0)
}

## ------------------------------------------------------- Wolfe line search
line_search <- function(f, g, x, d, a0 = 1, c1 = 1e-4, c2 = 0.9) {
  f0 <- f(x); gd0 <- sum(g(x) * d)
  if (!is.finite(gd0) || gd0 >= 0) return(list(alpha = 0, ok = FALSE))
  lo <- 0; hi <- Inf; a <- a0
  for (it in 1:60) {
    fa <- f(x + a * d)
    if (!is.finite(fa) || fa > f0 + c1 * a * gd0) {
      hi <- a; a <- 0.5 * (lo + hi)
    } else {
      if (sum(g(x + a * d) * d) < c2 * gd0) {
        lo <- a
        a <- if (is.finite(hi)) 0.5 * (lo + hi) else min(2 * a, 1e4)
      } else {
        return(list(alpha = a, ok = TRUE))
      }
    }
    if (a < 1e-14) return(list(alpha = 0, ok = FALSE))
  }
  list(alpha = a, ok = (a > 0))
}

## ------------------------------------------------------------- algorithms
## the methods that step through mini-batches: one iteration is one epoch
MINIBATCH <- c("sgdls", "mom", "adam")
## Row k of P, G, F is the state after k - 1 steps; A[k] is the step length
## that led to it.
run_alg <- function(alg, th0, x, y, maxit, lr = 0.1, mom = 0.9, m = 20,
                    tol = 1e-6) {
  p <- length(th0)
  f <- function(th) loss_fun(th, x, y)
  g <- function(th) loss_grad(th, x, y)
  P <- G <- matrix(NA_real_, maxit + 1, p)
  Fv <- AL <- rep(NA_real_, maxit + 1)
  th <- th0; B <- diag(p)
  g_prev <- d_prev <- s_prev <- NULL
  n <- 0; note <- "maximum number of iterations reached"
  ## state for the fixed-rate rules
  n_obs <- length(x); m <- max(1, min(m, n_obs))
  capped <- FALSE
  if (alg %in% MINIBATCH) {                  # keep the number of mini-batch steps in check
    cap <- max(1, floor(20000 / ceiling(n_obs / m)))
    if (maxit > cap) { maxit <- cap; capped <- TRUE }
  }
  vel <- m1 <- m2 <- numeric(p)              # momentum velocity; Adam moments
  tt <- 0                                    # mini-batch updates so far (Adam)
  if (alg %in% MINIBATCH) set.seed(11)

  for (k in 1:(maxit + 1)) {
    gk <- g(th); fk <- f(th)
    P[k, ] <- th; G[k, ] <- gk; Fv[k] <- fk; n <- k
    if (!is.finite(fk) || !all(is.finite(gk))) { note <- "diverged"; break }
    if (sqrt(sum(gk^2)) < tol) { note <- "converged: gradient norm < 1e-6"; break }
    if (k > maxit) break

    if (alg == "gdfix") {                    # full-batch gradient, fixed learning rate
      AL[k + 1] <- lr
      th <- th - lr * gk
      next
    }

    if (alg %in% MINIBATCH) {                # one iteration is one epoch: a shuffled pass
      perm <- sample.int(n_obs)              # of mini-batch steps over all the data, so
      al <- numeric(0)                       # it costs about one full-batch gradient
      for (s0 in seq(1, n_obs, by = m)) {
        idx <- perm[s0:min(s0 + m - 1, n_obs)]
        xb <- x[idx]; yb <- y[idx]
        gb <- loss_grad(th, xb, yb)               # the gradient of mini-batch B_k
        if (alg == "sgdls") {                # backtracking line search on the batch loss
          fb <- loss_fun(th, xb, yb); a <- lr; gg <- sum(gb^2)
          for (j in 1:30) {
            fa <- loss_fun(th - a * gb, xb, yb)
            if (is.finite(fa) && fa <= fb - 1e-4 * a * gg) break
            a <- a / 2
          }
          al <- c(al, a)
          th <- th - a * gb
        } else if (alg == "mom") {           # heavy ball on the batch gradient
          vel <- mom * vel - lr * gb
          th <- th + vel
        } else {                             # Adam on the batch gradient
          tt <- tt + 1
          m1 <- 0.9 * m1 + 0.1 * gb
          m2 <- 0.999 * m2 + 0.001 * gb^2
          th <- th - lr * (m1 / (1 - 0.9^tt)) / (sqrt(m2 / (1 - 0.999^tt)) + 1e-8)
        }
      }
      AL[k + 1] <- if (alg == "sgdls") mean(al) else lr
      next
    }

    if (alg == "gd") {
      d <- -gk
    } else if (alg == "cg") {
      if (is.null(d_prev)) {
        d <- -gk
      } else {
        yk <- gk - g_prev
        beta <- max(0, sum(gk * yk) / max(sum(g_prev^2), 1e-300))   # PR+
        d <- -gk + beta * d_prev
      }
      if (sum(gk * d) > -1e-12) d <- -gk                           # restart
    } else {                                                       # BFGS
      if (!is.null(s_prev)) {
        yk <- gk - g_prev; sy <- sum(s_prev * yk)
        if (sy > 1e-10) {
          if (k == 2) B <- (sy / sum(yk^2)) * diag(p)    # scale the first B
          rho <- 1 / sy
          V <- diag(p) - rho * outer(s_prev, yk)
          B <- V %*% B %*% t(V) + rho * outer(s_prev, s_prev)
        }
      }
      d <- -as.vector(B %*% gk)
      if (sum(gk * d) > -1e-12) { B <- diag(p); d <- -gk }
    }

    a0 <- if (alg == "bfgs" && k > 1) 1 else min(1, 1 / sqrt(sum(d^2)))
    ls <- line_search(f, g, th, d, a0)
    if (!ls$ok) { note <- "line search failed"; break }
    AL[k + 1] <- ls$alpha
    s_prev <- ls$alpha * d; g_prev <- gk; d_prev <- d
    th <- th + s_prev
  }
  if (capped && grepl("^maximum", note))
    note <- paste0("stopped after ", maxit, " epochs (20000 mini-batch steps)")
  i <- seq_len(n)
  list(P = P[i, , drop = FALSE], G = G[i, , drop = FALSE], F = Fv[i],
       A = AL[i], n = n, note = note)
}

ALGS <- c("Gradient descent, fixed learning rate"     = "gdfix",
          "Gradient descent + line search"            = "gd",
          "Conjugate gradient (Polak-Ribiere+)"       = "cg",
          "BFGS (quasi-Newton)"                       = "bfgs",
          "SGD + line search (on each mini-batch)"    = "sgdls",
          "SGD + momentum"                            = "mom",
          "SGD + Adam"                                = "adam")

## --------------------------------------------------------- network layout
## Weights sit on the edges; each intercept is a node of its own, a box with
## the value and its arrows, hung on the neuron it shifts.
NX <- c(0.07, 0.50)                                 # (x, y) of the input node
YH <- seq(0.88, 0.12, length.out = J); XH <- 0.60   # hidden nodes
NO <- c(1.13, 0.50)                                 # output node
at <- function(p, q, f) p + f * (q - p)             # a point along an edge
## hidden intercepts hang above the upper half of the hidden nodes and below the rest
YB <- YH + rep(c(0.14, -0.14), each = J / 2)
BIAS <- c(rep(0.56, J), 1.00)                       # x of the intercept boxes
BIASY <- c(YB, 0.17)                                # y of the intercept boxes
## where each weight's value and arrows sit
ADJ <- local({
  hj <- lapply(1:J, function(j) c(XH, YH[j]))
  pos <- c(lapply(1:J, function(j) at(NX, hj[[j]], 0.50)),    # w_j1: x -> h_j
           lapply(1:J, function(j) c(BIAS[j], BIASY[j])),     # w_j0: box on h_j
           lapply(1:J, function(j) at(hj[[j]], NO, 0.50)),    # v_j: h_j -> out
           list(c(BIAS[J + 1], BIASY[J + 1])))                # v_0: box on output
  data.frame(idx = 1:NP, sym = NAMES,
             x = sapply(pos, `[`, 1), y = sapply(pos, `[`, 2),
             stringsAsFactors = FALSE)
})
## the arrows beside each value (also the click targets): up / down beside an
## edge weight, left / right beside an intercept box
AX <- 0.104                                  # arrow offset from the value
isb <- ADJ$idx %in% BIASIDX
ARROWS <- rbind(
  data.frame(idx = ADJ$idx, dir = +1, horiz = isb,
             ax = ifelse(isb, ADJ$x + AX, ADJ$x + AX),
             ay = ifelse(isb, ADJ$y, ADJ$y + 0.028)),
  data.frame(idx = ADJ$idx, dir = -1, horiz = isb,
             ax = ifelse(isb, ADJ$x - AX, ADJ$x + AX),
             ay = ifelse(isb, ADJ$y, ADJ$y - 0.028)))

## compander: maps v in [-vmax, vmax] to [-1, 1], boosting small values
tone <- function(v, vmax, k) {
  u <- pmax(-1, pmin(1, v / vmax))
  sign(u) * log1p(k * abs(u)) / log1p(k)
}
hcol <- function(v, vmax, k) DIV[round((tone(v, vmax, k) + 1) * 100) + 1]

draw_net <- function(v, vmax, k, main, arrows = TRUE, digits = 2) {
  col <- hcol(v, vmax, k); lw <- 0.8 + 6 * abs(tone(v, vmax, k))
  plot.new(); plot.window(c(0, 1.2), c(-0.25, 1.08))
  title(main = main, cex.main = 1.05)
  H <- cbind(XH, YH)
  ## edges, each carrying one weight: x -> h_j (w_j1), h_j -> output (v_j)
  for (j in 1:J) {
    segments(NX[1], NX[2], H[j, 1], H[j, 2], col = col[j],     lwd = lw[j],     lend = 1)
    segments(H[j, 1], H[j, 2], NO[1], NO[2], col = col[2 * J + j], lwd = lw[2 * J + j], lend = 1)
  }
  ## intercept nodes: a thin arrow from each box to the neuron it shifts
  tgt <- rbind(H, NO)
  for (i in seq_along(BIASIDX)) {
    b <- ADJ[BIASIDX[i], ]
    dy <- sign(tgt[i, 2] - b$y)
    if (i == J + 1) arrows(b$x, b$y + 0.03, NO[1] - 0.03, NO[2] - 0.03, length = 0.08,
                       col = "grey45", lwd = 1.4)
    else arrows(b$x, b$y + dy * 0.035, tgt[i, 1], tgt[i, 2] - dy * 0.035,
                length = 0.08, col = "grey45", lwd = 1.4)
  }
  ## nodes
  points(rbind(NX, H, NO), pch = 21, cex = 3.2, bg = "white", lwd = 1.6)
  text(NX[1], NX[2], "x", font = 2)
  text(H[, 1], H[, 2], paste0("h", 1:J), cex = 0.8)
  text(NO[1], NO[2], expression(mu), cex = 1.1)
  ## values: an intercept is a box coloured by its value; an edge weight is
  ## plain text on a white patch so the edge does not strike through it
  for (i in 1:NP) {
    b <- ADJ[i, ]; isb <- i %in% BIASIDX
    if (isb) rect(b$x - 0.075, b$y - 0.036, b$x + 0.075, b$y + 0.036,
                  col = col[i], border = "grey35")
    else rect(b$x - 0.072, b$y - 0.028, b$x + 0.072, b$y + 0.028,
              col = adjustcolor("white", 0.8), border = NA)
    dark <- abs(tone(v[i], vmax, k)) > 0.62
    text(b$x, b$y, sprintf(paste0("%s=%.", digits, "f"), b$sym, v[i]),
         cex = 0.85, font = if (isb) 1 else 2,
         col = if (isb) { if (dark) "white" else "black" }
               else if (dark) col[i] else "grey20")
  }
  if (arrows) for (r in seq_len(nrow(ARROWS))) {
    a <- ARROWS[r, ]
    if (a$horiz)                                   # left / right: -, +
      polygon(a$ax + a$dir * c(-0.013, -0.013, 0.015),
              a$ay + c(-0.033, 0.033, 0), col = "grey30", border = NA)
    else                                           # down / up: -, +
      polygon(a$ax + c(-0.026, 0.026, 0), a$ay + a$dir * c(-0.016, -0.016, 0.02),
              col = "grey30", border = NA)
  }
  ## signed colour bar
  xs <- seq(0.12, 1.08, length.out = 201)
  rect(xs, -0.24, xs + diff(xs)[1], -0.19, col = DIV, border = NA)
  lab <- formatC(vmax, format = "g", digits = 2)
  text(c(0.12, 0.60, 1.08), -0.16, c(paste0("-", lab), "0", paste0("+", lab)),
       cex = 0.72)
}

## ----------------------------------------------------------------- server
server <- function(input, output, session) {

  dseed <- reactiveVal(1); rseed <- reactiveVal(1)
  theta0 <- reactiveVal(rand_start(1))        # the starting weights

  observeEvent(input$newdata, dseed(dseed() + 1))
  ## a sensible starting learning rate for each fixed-rate method
  for (s in c("A", "B")) local({
    s <- s
    observeEvent(input[[paste0("alg", s)]], {
      lr0 <- c(gdfix = 0.3, sgdls = 0.3, mom = 0.03, adam = 0.1)[input[[paste0("alg", s)]]]
      if (!is.na(lr0)) updateSliderInput(session, paste0("lr", s), value = unname(lr0))
    })
  })
  observeEvent(input$setzero, theta0(rep(0, NP)))
  observeEvent(input$newstart, {
    rseed(rseed() + 1); theta0(rand_start(rseed()))
  })
  ## an arrow click moves the nearest arrow's weight by one step
  observeEvent(input$netv_click, {
    cl <- input$netv_click
    d <- ((cl$x - ARROWS$ax) / 0.05)^2 + ((cl$y - ARROWS$ay) / 0.04)^2
    i <- which.min(d)
    if (d[i] < 2) {
      th <- theta0()
      th[ARROWS$idx[i]] <- th[ARROWS$idx[i]] + ARROWS$dir[i] * as.numeric(input$astep)
      theta0(th)
    }
  })

  ## the predictor on a different scale: x -> s x, with the slope weights of the generating
  ## network divided by s, so the true curve is the same function of x / s
  xsc <- reactive(if (is.null(input$xscale)) 1 else as.numeric(input$xscale))
  TT <- reactive({ th <- TRUE_THETA; th[1:J] <- th[1:J] / xsc(); th })
  observeEvent(input$settrue, theta0(TT()))
  DAT <- reactive({
    D <- make_data(input$n, as.numeric(input$sigma), dseed())
    D$x <- D$x * xsc()
    D
  })
  ## one run for each optimizer, from the same start on the same data
  mkrun <- function(s) reactive({
    D <- DAT(); a <- input[[paste0("alg", s)]]
    if (is.null(a) || a == "none") return(NULL)
    fixed <- a %in% c("gdfix", "sgdls", "mom", "adam")
    req(!fixed || !is.null(input[[paste0("lr", s)]]))
    run_alg(a, theta0(), D$x, D$y, as.numeric(input$maxit),
            lr  = if (fixed) input[[paste0("lr", s)]] else NA,
            mom = if (a == "mom") input[[paste0("mom", s)]] else NA,
            m   = if (a %in% MINIBATCH) as.numeric(input[[paste0("batch", s)]]) else NA)
  })
  runA <- mkrun("A"); runB <- mkrun("B")
  ## the network panels show the run chosen in "Network of"
  netsel <- reactive(if (identical(input$netsel, "B") && !is.null(runB())) "B" else "A")
  run <- reactive(if (netsel() == "B") runB() else runA())

  ## the browser owns the iteration (slider, play timer, buttons); a new run restarts it
  observeEvent(list(runA(), runB()), {
    nmax <- max(runA()$n, if (is.null(runB())) 0L else runB()$n)
    session$sendCustomMessage("krun", list(max = nmax - 1L,
                                           play = isTRUE(input$autoplay) && nmax > 1))
  })
  kraw <- reactive(as.integer(if (is.null(input$kshow)) 0L else input$kshow))
  kpos <- reactive(min(kraw(), run()$n - 1L))

  cur <- reactive({
    R <- run(); k <- min(kpos(), R$n - 1L) + 1L
    list(R = R, k = k, th = R$P[k, ], g = R$G[k, ])
  })

  output$netv <- renderPlot({
    C <- cur(); R <- C$R
    vmax <- max(3, quantile(abs(R$P), 0.95))
    par(mar = c(1, 1, 3, 1))
    draw_net(C$th, vmax, as.numeric(input$ctr),
             sprintf("%s: weights, iteration %d   (loss %.4f)", netsel(), C$k - 1L, R$F[C$k]))
  })

  output$netg <- renderPlot({
    C <- cur()
    gmax <- max(1e-8, max(abs(C$g)))
    par(mar = c(1, 1, 3, 1))
    draw_net(C$g, gmax, as.numeric(input$ctr),
             sprintf("%s: gradient of the loss, iteration %d", netsel(), C$k - 1L),
             arrows = FALSE, digits = 3)
  })

  ## one clearly different colour per hidden unit
  PCOL <- c("#F28E00", "#1F5FBF", "black", "#00B5CC")

  output$fit <- renderPlot({
    RA <- runA(); RB <- runB(); R <- RA; D <- DAT(); kk <- kraw()
    xg <- seq(-3, 3, length.out = 300) * xsc()
    par(mar = c(4, 4.2, 2.5, 1))
    plot(D$x, D$y, pch = 19, cex = 0.7, col = adjustcolor("#E8820C", 0.5),
         ylim = range(D$y, mu_fun(TT(), xg)),
         xlab = "x", ylab = expression(mu(x) ~ "and data y"),
         main = "fitted mean and the generated data")
    lines(xg, mu_fun(TT(), xg), col = "red", lty = 2, lwd = 2)
    lines(xg, mu_fun(R$P[1, ], xg), col = "grey60", lwd = 2)
    lines(xg, mu_fun(RA$P[min(kk, RA$n - 1L) + 1L, ], xg), col = "#2E8B57", lwd = 3)
    if (!is.null(RB))
      lines(xg, mu_fun(RB$P[min(kk, RB$n - 1L) + 1L, ], xg), col = "#1F5FBF", lwd = 2.5)
    leg <- data.frame(
      lab = c("true mean", "starting weights", paste("A:", alg_name(input$algA)), "data"),
      lwd = c(2, 2, 3, NA), lty = c(2, 1, 1, NA), pch = c(NA, NA, NA, 19),
      col = c("red", "grey60", "#2E8B57", adjustcolor("#E8820C", 0.7)),
      stringsAsFactors = FALSE)
    if (!is.null(RB))
      leg <- rbind(leg, data.frame(lab = paste("B:", alg_name(input$algB)), lwd = 2.5, lty = 1,
                                   pch = NA, col = "#1F5FBF", stringsAsFactors = FALSE))
    legend("topright", bty = "n", cex = 0.78, lwd = leg$lwd, lty = leg$lty,
           pch = leg$pch, col = leg$col, legend = leg$lab)
  })

  ## the scaled units, which add up to the fit on the same scale as the data
  output$parts <- renderPlot({
    C <- cur(); xg <- seq(-3, 3, length.out = 300) * xsc()
    VH <- unit_terms(C$th, xg)
    fit <- mu_fun(C$th, xg)
    mu_true <- mu_fun(TT(), xg)
    par(mar = c(4, 4.2, 2.5, 1))
    matplot(xg, VH, type = "l", lty = 3, lwd = 2.5, col = PCOL,
            ylim = pmax(-8, pmin(8, range(VH, fit, mu_true))),
            xlab = "x", ylab = expression(mu(x)),
            main = "the scaled units v_j h_j add up to the fit (with v0)")
    abline(h = 0, lty = 3, col = "grey70")
    lines(xg, mu_true, col = "red", lty = 2, lwd = 2)
    lines(xg, fit, col = "#2E8B57", lwd = 3)
    legend("topright", bty = "n", cex = 0.78, ncol = 2,
           lwd = c(rep(2.5, J), 3, 2), lty = c(rep(3, J), 1, 2),
           col = c(PCOL, "#2E8B57", "red"),
           legend = c(paste0("v", 1:J, " h", 1:J), "v0 + sum (fit)", "true"))
  })

  output$conv <- renderPlot({
    R <- runA(); RF <- runB(); C <- list(R = R, k = min(kraw(), R$n - 1L) + 1L)
    it <- 0:(R$n - 1)
    gn <- sqrt(rowSums(R$G^2))
    par(mfrow = c(1, 2), mar = c(4, 4.2, 2.5, 1))
    xl <- c(0, max(R$n, if (is.null(RF)) 0 else RF$n) - 1)
    ## the loss on the scale chosen below: as is, its log10, or the log10 of its excess
    ## over the smallest value either run reaches (so differences near the floor show)
    sc <- if (is.null(input$lscale)) "log" else input$lscale
    best <- min(c(R$F, RF$F), na.rm = TRUE)
    ls_fun <- function(v) switch(sc, lin = v, log = log10(pmax(v, 1e-12)),
                                 exc = log10(pmax(v - best, 1e-12)))
    ls_lab <- switch(sc, lin = "loss", log = expression(log[10] ~ "loss"),
                     exc = expression(log[10] ~ "(loss - smallest reached)"))
    plot(it, ls_fun(R$F), type = "b", pch = 20, col = "#2E8B57", xlab = "iteration",
         ylab = ls_lab, xlim = xl, ylim = range(c(ls_fun(R$F), ls_fun(RF$F)), finite = TRUE),
         main = "loss (half the mean squared error)")
    if (sc != "exc") abline(h = ls_fun(0.5 * DAT()$sigma^2), lty = 2, col = "grey50")  # the noise floor
    if (!is.null(RF)) {
      lines(0:(RF$n - 1), ls_fun(RF$F), type = "b", pch = 20, col = "#1F5FBF")
      kb <- min(kraw(), RF$n - 1L)
      points(kb, ls_fun(RF$F[kb + 1L]), pch = 19, col = "#1F5FBF", cex = 1.6)
      legend("topright", bty = "n", cex = 0.8, lty = 1, pch = 20,
             col = c("#2E8B57", "#1F5FBF"),
             legend = c(paste("A:", alg_name(input$algA)), paste("B:", alg_name(input$algB))))
    }
    points(it[C$k], ls_fun(R$F[C$k]), pch = 19, col = "#2E8B57", cex = 1.6)
    gnr <- if (is.null(RF)) NULL else log10(sqrt(rowSums(RF$G^2)))
    plot(it, log10(gn), type = "b", pch = 20, col = "#2E8B57", xlab = "iteration",
         xlim = xl, ylim = range(c(log10(gn), gnr), finite = TRUE),
         ylab = expression(log[10] ~ "||gradient||"), main = "gradient norm")
    if (!is.null(RF)) {
      lines(0:(RF$n - 1), gnr, type = "b", pch = 20, col = "#1F5FBF")
      points(kb, gnr[kb + 1L], pch = 19, col = "#1F5FBF", cex = 1.6)
    }
    points(it[C$k], log10(gn[C$k]), pch = 19, col = "#2E8B57", cex = 1.6)
  })

  output$info <- renderUI({
    D <- DAT(); kk <- kraw()
    line <- function(R, a, tag, col) {
      k <- min(kk, R$n - 1L) + 1L; al <- R$A[k]
      paste0("<b style='color:", col, "'>", tag, ": ", alg_name(a), "</b> &nbsp; <b>",
             if (a %in% MINIBATCH) "epoch" else "iteration", "</b> ", k - 1L, " of ", R$n - 1L,
             " &nbsp; <b>loss</b> ", formatC(R$F[k], digits = 5, format = "f"),
             " &nbsp; <b>noise floor</b> ", formatC(0.5 * D$sigma^2, digits = 4, format = "f"),
             " &nbsp; <b>||gradient||</b> ", formatC(sqrt(sum(R$G[k, ]^2)), digits = 3, format = "g"),
             if (is.finite(al)) paste0(" &nbsp; <b>",
                                       if (a %in% c("gdfix", "mom", "adam")) "learning rate"
                                       else "step length", "</b> ",
                                       formatC(al, digits = 3, format = "g")) else "",
             "<br><i>", R$note, "</i>")
    }
    HTML(paste(line(runA(), input$algA, "A", "#2E8B57"),
               if (!is.null(runB())) line(runB(), input$algB, "B", "#1F5FBF"),
               sep = "<br>"))
  })
}
