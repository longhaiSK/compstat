### renderPlot replacement: Safari 27 (WebKit) cannot hold the call stack of
### shiny::renderPlot in webR ("Maximum call stack size exceeded"), so draw with
### renderImage instead, which nests about 45 fewer R calls
renderPlot <- function(expr, width = "auto", height = "auto", res = 72, ...,
                       env = parent.frame(), quoted = FALSE) {
    func <- shiny::exprToFunction(expr, env, quoted)
    shiny::renderImage({
        info <- shiny::getCurrentOutputInfo()
        w  <- if (is.numeric(width))  width  else info$width()
        h  <- if (is.numeric(height)) height else info$height()
        shiny::req(w > 0, h > 0)
        pr <- shiny::getDefaultReactiveDomain()$clientData$pixelratio
        if (is.null(pr)) pr <- 1
        f   <- tempfile(fileext = ".png")
        dev <- shiny:::startPNG(f, w * pr, h * pr, res = res * pr)
        on.exit(grDevices::dev.off(dev), add = TRUE)
        grDevices::dev.control(displaylist = "enable")
        v <- withVisible(func())
        if (v$visible) print(v$value)
        list(src = f, contentType = "image/png", width = w, height = h,
             coordmap = shiny:::getCoordmap(v$value, w * pr, h * pr, res * pr))
    }, deleteFile = TRUE)
}

log_sum_exp <- function(lx) { m <- max(lx); m + log(sum(exp(lx - m))) }

sigma_prior <- 10                                       # prior: theta ~ N(0, 10^2)
grid  <- seq(-60, 60, by = 0.01)                        # fine grid for the exact answers and the modes
xlim0 <- c(-20, 20)                                     # fixed x-axis of the plots

### data: two clusters of Cauchy(theta, 1) observations at -sep and +sep
make_data <- function(sep, n_left, n_right) {
    spread <- function(n) if (n == 1) 0 else seq(-0.4, 0.4, length.out = n)
    c(-sep + spread(n_left), sep + spread(n_right))
}

### log of the unnormalized posterior kernel f(theta) = prod_i Cauchy(y_i | theta, 1) * N(theta | 0, 10^2)
log_f <- function(theta, y) {
    ll <- 0
    for (yi in y) ll <- ll + dcauchy(yi, theta, 1, log = TRUE)
    ll + dnorm(theta, 0, sigma_prior, log = TRUE)
}

### all the local modes of f found on the grid, each with the Laplace sd 1 / sqrt(-(log f)'')
find_modes <- function(y) {
    lf <- log_f(grid, y); h <- grid[2] - grid[1]
    i  <- which(lf[-c(1, length(lf))] > lf[-c(length(lf) - 1, length(lf))] &
                lf[-c(1, length(lf))] > lf[-c(1, 2)]) + 1
    i  <- i[lf[i] > max(lf) - 30]
    d2 <- (lf[i + 1] - 2 * lf[i] + lf[i - 1]) / h^2
    data.frame(mode = grid[i], logf = lf[i], sd = 1 / sqrt(-d2))
}

server <- function(input, output, session) {

    setup <- reactive({
        y  <- make_data(input$sep, input$n_left, input$n_right)
        lf <- log_f(grid, y); h <- grid[2] - grid[1]
        ### exact answers by a fine grid: the normalizing constant Z = P(y) and the posterior mean
        log_Z  <- log_sum_exp(lf) + log(h)
        post_m <- sum(exp(lf - max(lf)) * grid) / sum(exp(lf - max(lf)))
        modes  <- find_modes(y)
        list(y = y, log_Z = log_Z, post_m = post_m, modes = modes)
    })

    ### the proposal g: a normal at the mode nearest the start value, or a mixture of normals at all modes
    proposal <- function(s) {
        m <- s$modes
        if (input$prop == "one") {
            k <- which.min(abs(m$mode - input$start))
            m <- m[k, , drop = FALSE]
        }
        pi_k <- exp(m$logf - max(m$logf)) * m$sd            # Laplace mass of each mode
        pi_k <- pi_k / sum(pi_k)
        sd_k <- m$sd * input$infl
        list(pi = pi_k, mu = m$mode, sd = sd_k,
             log_g = function(x) {
                 lg <- vapply(seq_along(pi_k), function(k) log(pi_k[k]) + dnorm(x, m$mode[k], sd_k[k], log = TRUE),
                              numeric(length(x)))
                 if (length(x) == 1) lg <- matrix(lg, 1)
                 apply(lg, 1, log_sum_exp)
             },
             draw = function(S) {
                 k <- sample.int(length(pi_k), S, replace = TRUE, prob = pi_k)
                 rnorm(S, m$mode[k], sd_k[k])
             })
    }

    sim <- reactive({
        s <- setup(); g <- proposal(s); S <- input$S
        set.seed(input$seed + 1000 * input$new)
        th   <- g$draw(S)
        logw <- log_f(th, s$y) - g$log_g(th)
        mx   <- max(logw); v <- exp(logw - mx); wn <- v / sum(v)
        log_Zhat <- mx + log(mean(v))
        rel_se   <- sd(v) / (sqrt(S) * mean(v))
        list(s = s, g = g, th = th, wn = wn, S = S, log_Zhat = log_Zhat, rel_se = rel_se,
             Ehat = sum(wn * th), ess = 1 / sum(wn^2), lsum = mx + log(sum(v)))
    })

    output$panels <- renderPlot({
        z <- sim(); s <- z$s; th <- z$th
        xs <- seq(xlim0[1], xlim0[2], length.out = 1500)
        lfx <- log_f(xs, s$y)
        fx <- exp(lfx - s$log_Z)                            # posterior density f / Z
        gx <- exp(z$g$log_g(xs))
        lwc <- lfx - z$g$log_g(xs) - z$lsum

        layout(matrix(1:2, 2), heights = c(3, 2))
        par(mar = c(0.4, 5, 3, 1))
        plot(xs, fx, type = "n", ylim = c(0, max(fx, gx) * 1.1), xlim = xlim0, xaxt = "n",
             xlab = "", ylab = "density",
             main = "Cauchy-location posterior and the importance proposal")
        lines(xs, fx, lwd = 4, col = "red")
        lines(xs, gx, lwd = 3, col = "grey60")
        rug(th[th > xlim0[1] & th < xlim0[2]], col = adjustcolor("grey45", 0.35))
        points(s$modes$mode, rep(0, nrow(s$modes)), pch = 17, col = "red", cex = 1.3)
        legend("topright", bty = "n", cex = 1.05, text.col = "navy",
               lwd = c(4, 3, NA), pch = c(NA, NA, 17), col = c("red", "grey60", "red"),
               legend = c("posterior f / Z", "proposal g", "modes of the posterior"))

        par(mar = c(4.5, 5, 0.4, 1))
        wn <- z$wn
        wl <- c(min(wn) / 5, max(wn) * 5)
        plot(th, wn, type = "n", log = "y", ylim = wl, xlim = xlim0,
             xlab = expression(theta), ylab = "normalized weight (log scale)")
        lines(xs, exp(pmin(lwc, 700)), lwd = 2, col = "grey25")
        segments(th, wl[1], th, wn, col = adjustcolor("grey45", 0.5))
        abline(h = 1 / z$S, lty = 2, col = "grey55", lwd = 2)
        legend("bottomright", bty = "n", cex = 1.05, text.col = "navy",
               lty = c(2, 1), lwd = c(2, 2), col = c("grey55", "grey25"),
               legend = c("1/S", "normalized weight function"))
    })

    output$tab <- renderTable({
        z <- sim(); s <- z$s
        lo <- z$log_Zhat + log(max(1 - 1.96 * z$rel_se, 1e-12)); hi <- z$log_Zhat + log1p(1.96 * z$rel_se)
        data.frame(
            Quantity = c("log P(y)", "95% CI for log P(y)", "posterior mean", "observed ESS (% of S)"),
            True = c(sprintf("%.4f", s$log_Z), "", sprintf("%.4f", round(s$post_m, 4) + 0), ""),
            Estimate = c(sprintf("%.4f", z$log_Zhat), sprintf("(%.4f, %.4f)", lo, hi),
                         sprintf("%.4f", z$Ehat), sprintf("%.4g", 100 * z$ess / z$S)))
    }, striped = TRUE, align = "lrr", width = "100%")

    output$modes <- renderTable({
        s <- setup(); m <- s$modes
        data.frame(mode = sprintf("%.2f", m$mode),
                   `Laplace sd` = sprintf("%.3f", m$sd),
                   `share of the mass` = sprintf("%.3f", {
                       p <- exp(m$logf - max(m$logf)) * m$sd; p / sum(p) }),
                   check.names = FALSE)
    }, striped = TRUE, align = "lrr", width = "100%")
}

shinyApp(ui, server)

About the app

Shows that importance sampling can miss a whole mode of a realistic posterior, the location of a Cauchy sample, while every diagnostic looks fine. It is a realistic version of the failure in The Failure of Importance Sampling.

The model. The data are \(y_1,\ldots,y_n\overset{iid}{\sim}\text{Cauchy}(\theta,1)\) with the prior \(\theta\sim N(0,10^2)\). The data are arranged in two clusters, around \(-s\) and \(+s\) with \(s\) the “cluster separation”, with a chosen number of observations in each. The unnormalized posterior and its normalizing constant, the marginal likelihood, are \[f(\theta)=\prod_{i=1}^n \frac{1}{\pi\{1+(y_i-\theta)^2\}}\;N(\theta\mid0,10^2),\qquad Z=P(y)=\int f(\theta)\,d\theta .\] The Cauchy likelihood is not log-concave, so when the clusters are far apart \(f\) has several modes, shown as triangles in the top plot and listed in the table with their Laplace standard deviations and shares of the posterior mass. The exact \(Z\) and posterior mean in the table come from a fine grid.

The proposal. Two proposals are available. “Laplace at one mode” is the common practice: run an optimizer from the start value (slider), find the mode it converges to, \(\hat\theta\), and use \(g(\theta)=N(\theta\mid\hat\theta,\,c^2s_L^2)\) with \(s_L=\{-(\log f)''(\hat\theta)\}^{-1/2}\) the Laplace standard deviation and \(c\) the “proposal sd inflation”. “Mixture of Laplace at every mode” is the cure: \(g(\theta)=\sum_k\pi_kN(\theta\mid\hat\theta_k,\,c^2s_{L,k}^2)\) with \(\pi_k\propto f(\hat\theta_k)\,s_{L,k}\), the Laplace mass of mode \(k\).

The estimators. Draw \(\theta_1,\ldots,\theta_S\sim g\) and form the weights \(w_s=f(\theta_s)/g(\theta_s)\). Since \(E_g(w)=Z\), \[\hat Z=\frac1S\sum_{s=1}^S w_s,\qquad \widehat{E_f(\theta)}=\frac{\sum_s w_s\theta_s}{\sum_s w_s},\] and the 95% interval for \(Z\) is \(\hat Z\,(1\pm1.96\,s_w/(\bar w\sqrt S))\), shown on the log scale. The observed effective sample size is \(S_{\rm eff}=1/\sum_s\tilde w_s^2\) with \(\tilde w_s=w_s/\sum_jw_j\), reported as a percentage of \(S\).

The two plots. Top: the posterior density \(f/Z\) (red) and the proposal \(g\) (grey), with the draws as a rug at the bottom. Bottom: the normalized weights of the draws on a log scale, with the weight function \(f/g\) and the dashed line \(1/S\) of perfectly balanced weights.

What to try. Keep the default (two clusters of three observations at \(\pm5\)) and the proposal “Laplace at one mode”. The posterior has one mode in each cluster, but the proposal covers only the one found from the start value. The weights are nearly equal and the observed effective sample size is close to 100% of \(S\), so the diagnostics report a healthy run, but \(\log\hat Z\) is too small by \(-\log(\text{share of the covered mode})\) (about \(\log 2\) for two equal modes) and the posterior mean estimate sits in one cluster instead of near \(0\). The CI is narrow and wrong. Move the start value across the middle and the optimizer lands on the other mode, with the same failure. Now switch to the mixture of Laplace approximations: the estimate and the CI move onto the truth. Move the clusters closer (\(s\lesssim2\)) and the modes merge and the failure disappears; make the clusters unequal and the share of the missed mode shrinks and so does the error. The lesson is that a proposal built from a local approximation reports nothing about the modes it never visited; the check is to search for modes from many starting points before trusting an importance-sampling estimate.
