## ------------------------------------------------------------------- model
## One covariate x, four tanh hidden units, log link, Poisson counts:
##   a_j = w_j1 x + w_j0,  h_j = tanh(a_j),  eta = v_0 + sum_j v_j h_j,
##   lambda = exp(eta).
## theta = (w11..wJ1,  w10..wJ0,  v1..vJ,  v0)
J <- 4
NP <- 3 * J + 1                                  # number of weights
NAMES <- c(paste0("w", 1:J, "1"), paste0("w", 1:J, "0"), paste0("v", 1:J), "v0")
BIASIDX <- c(J + 1:J, NP)                        # the intercepts w_j0 and v_0
## two bumps in the mean: units 1, 2 switch on and off over (-2.2, -0.8),
## units 3, 4 over (0.8, 2.2), so lambda(x) has two modes
TRUE_THETA <- c(2.5, 2.5, 2.5, 2.5,   5.5, 2, -2, -5.5,   1.2, -1.2, 1.0, -1.0,   -0.5)

fwd <- function(th, x) {
  H <- tanh(outer(x, th[1:J]) + rep(th[J + 1:J], each = length(x)))
  list(H = H, eta = as.vector(th[3 * J + 1] + H %*% th[2 * J + 1:J]))
}
lambda_fun <- function(th, x) exp(fwd(th, x)$eta)

## the scaled units v_j h_j(x), one column per hidden unit
unit_terms <- function(th, x) sweep(fwd(th, x)$H, 2, th[2 * J + 1:J], "*")

## average Poisson negative log-likelihood (the log y! term dropped)
nll <- function(th, x, y) {
  e <- fwd(th, x)$eta
  v <- mean(exp(e) - y * e)
  if (is.finite(v)) v else Inf
}

## its gradient by backpropagation:  delta_eta -> delta_a -> weights
nll_grad <- function(th, x, y) {
  E <- fwd(th, x)
  d_eta <- exp(E$eta) - y                                   # n
  d_a   <- (1 - E$H^2) * outer(d_eta, th[2 * J + 1:J])      # n x J
  c(colSums(d_a * x),                                       # w_j1
    colSums(d_a),                                           # w_j0
    as.vector(crossprod(E$H, d_eta)),                       # v_j
    sum(d_eta)) / length(x)                                 # v_0
}

make_data <- function(n, seed) {
  set.seed(100 + seed)
  x <- runif(n, -3, 3)
  y <- rpois(n, lambda_fun(TRUE_THETA, x))
  ## a fixed vertical offset, drawn once with the data, so the points stay put
  list(x = x, y = y, yj = y + runif(n, -0.12, 0.12))
}

rand_start <- function(seed) {
  set.seed(2000 + seed)
  c(rnorm(3 * J, 0, 1), 0)
}

## ------------------------------------------------------- Wolfe line search
line_search <- function(f, g, x, d, a0 = 1, c1 = 1e-4, c2 = 0.9) {
  f0 <- f(x); gd0 <- sum(g(x) * d)
  if (!is.finite(gd0) || gd0 >= 0) return(list(alpha = 0, ok = FALSE))
  lo <- 0; hi <- Inf; a <- a0
  for (it in 1:60) {
    fa <- f(x + a * d)
    if (!is.finite(fa) || fa > f0 + c1 * a * gd0) {
      hi <- a; a <- 0.5 * (lo + hi)
    } else {
      if (sum(g(x + a * d) * d) < c2 * gd0) {
        lo <- a
        a <- if (is.finite(hi)) 0.5 * (lo + hi) else min(2 * a, 1e4)
      } else {
        return(list(alpha = a, ok = TRUE))
      }
    }
    if (a < 1e-14) return(list(alpha = 0, ok = FALSE))
  }
  list(alpha = a, ok = (a > 0))
}

## ------------------------------------------------------------- algorithms
## Row k of P, G, F is the state after k - 1 steps; A[k] is the step length
## that led to it.
run_alg <- function(alg, th0, x, y, maxit, lr = 0.1, mom = 0.9, m = 20,
                    tol = 1e-6) {
  p <- length(th0)
  f <- function(th) nll(th, x, y)
  g <- function(th) nll_grad(th, x, y)
  P <- G <- matrix(NA_real_, maxit + 1, p)
  Fv <- AL <- rep(NA_real_, maxit + 1)
  th <- th0; B <- diag(p)
  g_prev <- d_prev <- s_prev <- NULL
  n <- 0; note <- "maximum number of iterations reached"
  ## state for the fixed-rate rules
  n_obs <- length(x); m <- max(1, min(m, n_obs))
  capped <- FALSE
  if (alg == "sgd") {                        # keep the number of mini-batch steps in check
    cap <- max(1, floor(20000 / ceiling(n_obs / m)))
    if (maxit > cap) { maxit <- cap; capped <- TRUE }
  }
  vel <- m1 <- m2 <- numeric(p)              # momentum velocity; Adam moments
  if (alg == "sgd") set.seed(11)

  for (k in 1:(maxit + 1)) {
    gk <- g(th); fk <- f(th)
    P[k, ] <- th; G[k, ] <- gk; Fv[k] <- fk; n <- k
    if (!is.finite(fk) || !all(is.finite(gk))) { note <- "diverged"; break }
    if (sqrt(sum(gk^2)) < tol) { note <- "converged: gradient norm < 1e-6"; break }
    if (k > maxit) break

    if (alg %in% c("gdfix", "mom", "adam", "sgd")) {  # fixed rate, no line search
      if (alg == "sgd") {                    # one iteration is one epoch: a shuffled
        perm <- sample.int(n_obs)            # pass of mini-batch steps over all the
        for (s0 in seq(1, n_obs, by = m)) {  # data, so it is paired with a full-batch step
          idx <- perm[s0:min(s0 + m - 1, n_obs)]
          th <- th - lr * nll_grad(th, x[idx], y[idx])
        }
        AL[k + 1] <- lr
        next
      } else gs <- gk
      if (alg == "mom") {                    # heavy ball: keep a fraction of the last step
        vel <- mom * vel - lr * gs
        dth <- vel
      } else if (alg == "adam") {            # per-weight step from running moments
        m1 <- 0.9 * m1 + 0.1 * gs
        m2 <- 0.999 * m2 + 0.001 * gs^2
        dth <- -lr * (m1 / (1 - 0.9^k)) / (sqrt(m2 / (1 - 0.999^k)) + 1e-8)
      } else dth <- -lr * gs                 # gradient descent / SGD
      AL[k + 1] <- lr
      th <- th + dth
      next
    }

    if (alg == "gd") {
      d <- -gk
    } else if (alg == "cg") {
      if (is.null(d_prev)) {
        d <- -gk
      } else {
        yk <- gk - g_prev
        beta <- max(0, sum(gk * yk) / max(sum(g_prev^2), 1e-300))   # PR+
        d <- -gk + beta * d_prev
      }
      if (sum(gk * d) > -1e-12) d <- -gk                           # restart
    } else {                                                       # BFGS
      if (!is.null(s_prev)) {
        yk <- gk - g_prev; sy <- sum(s_prev * yk)
        if (sy > 1e-10) {
          if (k == 2) B <- (sy / sum(yk^2)) * diag(p)    # scale the first B
          rho <- 1 / sy
          V <- diag(p) - rho * outer(s_prev, yk)
          B <- V %*% B %*% t(V) + rho * outer(s_prev, s_prev)
        }
      }
      d <- -as.vector(B %*% gk)
      if (sum(gk * d) > -1e-12) { B <- diag(p); d <- -gk }
    }

    a0 <- if (alg == "bfgs" && k > 1) 1 else min(1, 1 / sqrt(sum(d^2)))
    ls <- line_search(f, g, th, d, a0)
    if (!ls$ok) { note <- "line search failed"; break }
    AL[k + 1] <- ls$alpha
    s_prev <- ls$alpha * d; g_prev <- gk; d_prev <- d
    th <- th + s_prev
  }
  if (capped && grepl("^maximum", note))
    note <- paste0("stopped after ", maxit, " epochs (20000 mini-batch steps)")
  i <- seq_len(n)
  list(P = P[i, , drop = FALSE], G = G[i, , drop = FALSE], F = Fv[i],
       A = AL[i], n = n, note = note)
}

ALGS <- c("Gradient descent, fixed learning rate" = "gdfix",
          "Gradient descent + line search"      = "gd",
          "SGD (mini-batch)"                    = "sgd",
          "Gradient descent + momentum"         = "mom",
          "Adam"                                = "adam",
          "Conjugate gradient (Polak-Ribiere+)" = "cg",
          "BFGS (quasi-Newton)"                 = "bfgs"
)

## --------------------------------------------------------- network layout
## Weights sit on the edges; each intercept is a node of its own, a box with
## the value and its arrows, hung on the neuron it shifts.
NX <- c(0.07, 0.50)                                 # (x, y) of the input node
YH <- seq(0.88, 0.12, length.out = J); XH <- 0.60   # hidden nodes
NO <- c(1.13, 0.50)                                 # output node
at <- function(p, q, f) p + f * (q - p)             # a point along an edge
## hidden intercepts hang above the upper half of the hidden nodes and below the rest
YB <- YH + rep(c(0.14, -0.14), each = J / 2)
BIAS <- c(rep(0.56, J), 1.00)                       # x of the intercept boxes
BIASY <- c(YB, 0.17)                                # y of the intercept boxes
## where each weight's value and arrows sit
ADJ <- local({
  hj <- lapply(1:J, function(j) c(XH, YH[j]))
  pos <- c(lapply(1:J, function(j) at(NX, hj[[j]], 0.50)),    # w_j1: x -> h_j
           lapply(1:J, function(j) c(BIAS[j], BIASY[j])),     # w_j0: box on h_j
           lapply(1:J, function(j) at(hj[[j]], NO, 0.50)),    # v_j: h_j -> out
           list(c(BIAS[J + 1], BIASY[J + 1])))                # v_0: box on output
  data.frame(idx = 1:NP, sym = NAMES,
             x = sapply(pos, `[`, 1), y = sapply(pos, `[`, 2),
             stringsAsFactors = FALSE)
})
## the arrows beside each value (also the click targets): up / down beside an
## edge weight, left / right beside an intercept box
AX <- 0.104                                  # arrow offset from the value
isb <- ADJ$idx %in% BIASIDX
ARROWS <- rbind(
  data.frame(idx = ADJ$idx, dir = +1, horiz = isb,
             ax = ifelse(isb, ADJ$x + AX, ADJ$x + AX),
             ay = ifelse(isb, ADJ$y, ADJ$y + 0.028)),
  data.frame(idx = ADJ$idx, dir = -1, horiz = isb,
             ax = ifelse(isb, ADJ$x - AX, ADJ$x + AX),
             ay = ifelse(isb, ADJ$y, ADJ$y - 0.028)))

## compander: maps v in [-vmax, vmax] to [-1, 1], boosting small values
tone <- function(v, vmax, k) {
  u <- pmax(-1, pmin(1, v / vmax))
  sign(u) * log1p(k * abs(u)) / log1p(k)
}
hcol <- function(v, vmax, k) DIV[round((tone(v, vmax, k) + 1) * 100) + 1]

draw_net <- function(v, vmax, k, main, arrows = TRUE, digits = 2) {
  col <- hcol(v, vmax, k); lw <- 0.8 + 6 * abs(tone(v, vmax, k))
  plot.new(); plot.window(c(0, 1.2), c(-0.25, 1.08))
  title(main = main, cex.main = 1.05)
  H <- cbind(XH, YH)
  ## edges, each carrying one weight: x -> h_j (w_j1), h_j -> output (v_j)
  for (j in 1:J) {
    segments(NX[1], NX[2], H[j, 1], H[j, 2], col = col[j],     lwd = lw[j],     lend = 1)
    segments(H[j, 1], H[j, 2], NO[1], NO[2], col = col[2 * J + j], lwd = lw[2 * J + j], lend = 1)
  }
  ## intercept nodes: a thin arrow from each box to the neuron it shifts
  tgt <- rbind(H, NO)
  for (i in seq_along(BIASIDX)) {
    b <- ADJ[BIASIDX[i], ]
    dy <- sign(tgt[i, 2] - b$y)
    if (i == J + 1) arrows(b$x, b$y + 0.03, NO[1] - 0.03, NO[2] - 0.03, length = 0.08,
                       col = "grey45", lwd = 1.4)
    else arrows(b$x, b$y + dy * 0.035, tgt[i, 1], tgt[i, 2] - dy * 0.035,
                length = 0.08, col = "grey45", lwd = 1.4)
  }
  ## nodes
  points(rbind(NX, H, NO), pch = 21, cex = 3.2, bg = "white", lwd = 1.6)
  text(NX[1], NX[2], "x", font = 2)
  text(H[, 1], H[, 2], paste0("h", 1:J), cex = 0.8)
  text(NO[1], NO[2], expression(lambda), cex = 1.1)
  ## values: an intercept is a box coloured by its value; an edge weight is
  ## plain text on a white patch so the edge does not strike through it
  for (i in 1:NP) {
    b <- ADJ[i, ]; isb <- i %in% BIASIDX
    if (isb) rect(b$x - 0.075, b$y - 0.036, b$x + 0.075, b$y + 0.036,
                  col = col[i], border = "grey35")
    else rect(b$x - 0.072, b$y - 0.028, b$x + 0.072, b$y + 0.028,
              col = adjustcolor("white", 0.8), border = NA)
    dark <- abs(tone(v[i], vmax, k)) > 0.62
    text(b$x, b$y, sprintf(paste0("%s=%.", digits, "f"), b$sym, v[i]),
         cex = 0.85, font = if (isb) 1 else 2,
         col = if (isb) { if (dark) "white" else "black" }
               else if (dark) col[i] else "grey20")
  }
  if (arrows) for (r in seq_len(nrow(ARROWS))) {
    a <- ARROWS[r, ]
    if (a$horiz)                                   # left / right: -, +
      polygon(a$ax + a$dir * c(-0.013, -0.013, 0.015),
              a$ay + c(-0.033, 0.033, 0), col = "grey30", border = NA)
    else                                           # down / up: -, +
      polygon(a$ax + c(-0.026, 0.026, 0), a$ay + a$dir * c(-0.016, -0.016, 0.02),
              col = "grey30", border = NA)
  }
  ## signed colour bar
  xs <- seq(0.12, 1.08, length.out = 201)
  rect(xs, -0.24, xs + diff(xs)[1], -0.19, col = DIV, border = NA)
  lab <- formatC(vmax, format = "g", digits = 2)
  text(c(0.12, 0.60, 1.08), -0.16, c(paste0("-", lab), "0", paste0("+", lab)),
       cex = 0.72)
}

## ----------------------------------------------------------------- server
server <- function(input, output, session) {

  dseed <- reactiveVal(1); rseed <- reactiveVal(1)
  theta0 <- reactiveVal(rand_start(1))        # the starting weights
  playing <- reactiveVal(FALSE)
  kpos <- reactiveVal(0L)                     # current iteration, 0 .. R$n - 1

  observeEvent(input$newdata, dseed(dseed() + 1))
  ## a sensible starting learning rate for each fixed-rate method
  observeEvent(input$alg, {
    lr0 <- c(gdfix = 0.1, sgd = 0.1, mom = 0.05, adam = 0.1)[input$alg]
    if (!is.na(lr0)) updateSliderInput(session, "lr", value = unname(lr0))
  })
  observeEvent(input$settrue, theta0(TRUE_THETA))
  observeEvent(input$setzero, theta0(rep(0, NP)))
  observeEvent(input$newstart, {
    rseed(rseed() + 1); theta0(rand_start(rseed()))
  })
  ## an arrow click moves the nearest arrow's weight by one step
  observeEvent(input$netv_click, {
    cl <- input$netv_click
    d <- ((cl$x - ARROWS$ax) / 0.05)^2 + ((cl$y - ARROWS$ay) / 0.04)^2
    i <- which.min(d)
    if (d[i] < 2) {
      th <- theta0()
      th[ARROWS$idx[i]] <- th[ARROWS$idx[i]] + ARROWS$dir[i] * as.numeric(input$astep)
      theta0(th)
    }
  })

  DAT <- reactive(make_data(input$n, dseed()))
  run <- reactive({
    D <- DAT()
    fixed <- input$alg %in% c("gdfix", "sgd", "mom", "adam")
    run_alg(input$alg, theta0(), D$x, D$y, as.numeric(input$maxit),
            lr  = if (fixed) input$lr else NA,
            mom = if (input$alg == "mom") input$mom else NA,
            m   = if (input$alg == "sgd") as.numeric(input$batch) else NA)
  })

  observeEvent(run(), {
    kpos(0L)
    updateSliderInput(session, "kshow", min = 0, max = run()$n - 1L, value = 0)
    playing(isTRUE(input$autoplay) && run()$n > 1)
  })

  ## the iteration slider follows the run, and scrubs it when dragged
  observeEvent(kpos(), updateSliderInput(session, "kshow", value = kpos()))
  observeEvent(input$kshow, {
    if (!isTRUE(playing()) && input$kshow != kpos())
      kpos(min(as.integer(input$kshow), run()$n - 1L))
  })

  observeEvent(input$play, {
    if (isTRUE(playing())) playing(FALSE)
    else {
      if (kpos() >= run()$n - 1) kpos(0L)
      playing(TRUE)
    }
  })
  observeEvent(playing(), {
    updateActionButton(session, "play",
                       icon = icon(if (isTRUE(playing())) "pause" else "play"))
  })
  observe({
    if (!isTRUE(playing())) return()
    kmax <- isolate(run()$n) - 1
    k <- isolate(kpos())
    if (k >= kmax) { playing(FALSE); return() }
    invalidateLater(300, session)
    kpos(k + 1L)
  })
  observeEvent(input$step,  { playing(FALSE); kpos(min(kpos() + 1L, run()$n - 1L)) })
  observeEvent(input$back,  { playing(FALSE); kpos(max(kpos() - 1L, 0L)) })
  observeEvent(input$first, { playing(FALSE); kpos(0L) })
  observeEvent(input$toend, { playing(FALSE); kpos(run()$n - 1L) })

  cur <- reactive({
    R <- run(); k <- min(kpos(), R$n - 1L) + 1L

About the app

Fits a Poisson neural network like the one in the book’s backpropagation section, with one covariate and four tanh hidden units, to counts generated from known weights, and lets you watch seven optimizers work on it: gradient descent with a fixed learning rate, SGD, momentum, Adam, gradient descent with a line search, conjugate gradient, and BFGS. The four units are enough for a mean with two modes: the true weights switch units 1 and 2 on and off around \(x \in (-2.2, -0.8)\), and units 3 and 4 around \((0.8, 2.2)\).

The weights. The upper network shows the thirteen weights at the current iteration. The eight weights on edges (w11, …, w41 from \(x\) to the hidden units and v1, …, v4 from the hidden units to the output) are written on their edges, and each intercept (w10, …, w40 and v0) is a node of its own, a box hung on the neuron it shifts. Colour shows the value (blue negative, red positive; on an edge it also sets the width). The arrows beside a value change that weight (up and down on an edge, right to increase and left to decrease on an intercept box), and the starting point of the run changes with it. True Value loads the weights that generated the data, Random draws a new set, and Zero sets every weight to 0 (a saddle point of the loss: with all hidden weights at zero, only \(v_0\) has a nonzero gradient). Adjust Size sets how much one click on an arrow changes a weight.

The methods. Write \(g_k = \nabla L(\theta_k)\) for the gradient of the average loss \(L\) over all \(n\) observations at the current weights, and \(\ell_i\) for the loss of observation \(i\). Every optimizer in the menu, in the order of the menu, can be written as one update

\[\theta_{k+1} = \theta_k - \alpha_k\, P_k\, m_k ,\]

where \(m_k\) is the gradient actually used (the gradient itself, a sampled version of it, or a smoothed version of it), \(P_k\) is a preconditioner that rescales or rotates the step, and \(\alpha_k\) is the step size.

Optimizer Gradient used, \(m_k\) Preconditioner, \(P_k\) Step size, \(\alpha_k\)
Fixed-rate GD \(g_k\) \(I\) \(\alpha\), set by hand
SGD \(g_k^{B} = \frac{1}{m}\sum_{i \in B_k} \nabla \ell_i(\theta_k)\) for a random mini-batch \(B_k\) of size \(m\), so that \(E\,g_k^{B} = g_k\) \(I\) \(\alpha\), set by hand
Momentum \(v_k = \mu\, v_{k-1} + g_k\), an exponentially weighted sum of past gradients \(I\) \(\alpha\), set by hand
Adam \(\hat m_k = \dfrac{(1-\beta_1)\sum_{j \le k} \beta_1^{k-j} g_j}{1-\beta_1^{k}}\), a bias-corrected average of past gradients \(\mathrm{diag}\big(1/(\sqrt{\hat v_k} + \epsilon)\big)\), with \(\hat v_k = \dfrac{(1-\beta_2)\sum_{j \le k} \beta_2^{k-j} g_j \odot g_j}{1-\beta_2^{k}}\) \(\alpha\), set by hand
GD + line search \(g_k\) \(I\) \(\alpha_k\) from a Wolfe line search
Conjugate gradient (PR+) \(d_k = -g_k + \beta_k d_{k-1}\), with \(\beta_k = \max\!\Big(0, \dfrac{g_k^\top (g_k - g_{k-1})}{g_{k-1}^\top g_{k-1}}\Big)\) implicit: the new gradient is combined with the previous direction \(\alpha_k\) from a Wolfe line search
BFGS \(g_k\) \(B_k \approx H^{-1}\), a full matrix, updated by \(B_{k+1} = (I - \rho\, s y^\top) B_k (I - \rho\, y s^\top) + \rho\, s s^\top\) with \(s = \theta_{k+1} - \theta_k\), \(y = g_{k+1} - g_k\), \(\rho = 1/(s^\top y)\) \(\alpha_k\) from a Wolfe line search

In the app \(\beta_1 = 0.9\), \(\beta_2 = 0.999\) and \(\epsilon = 10^{-8}\) for Adam. Momentum is coded as \(v \leftarrow \mu v - \alpha g\), \(\theta \leftarrow \theta + v\), the same update written with the sign and \(\alpha\) absorbed into \(v\). A Wolfe line search takes the step \(\alpha\) along \(d_k = -P_k m_k\) that satisfies both \(L(\theta_k + \alpha d_k) \le L(\theta_k) + c_1 \alpha\, g_k^\top d_k\) (sufficient decrease) and \(\nabla L(\theta_k + \alpha d_k)^\top d_k \ge c_2\, g_k^\top d_k\) (curvature), with \(c_1 = 10^{-4}\) and \(c_2 = 0.9\).

What each one changes. Fixed-rate GD, SGD and momentum do not change the direction’s shape at all: \(P_k = I\). SGD changes which gradient is used (a noisy but cheap one) and momentum changes it by averaging over the past, which damps the zig-zag across a valley and speeds up progress along it. Neither uses curvature or Fisher information. Adam is the first to precondition: dividing by \(\sqrt{\hat v_k}\) shrinks the step of weights with consistently large gradients and enlarges that of weights with small ones. For this model \(\nabla \ell_i = (\lambda_i - y_i)\nabla \eta_i\), and the empirical Fisher information is \(\hat F = \frac{1}{n}\sum_i (\lambda_i - y_i)^2 \nabla\eta_i \nabla\eta_i^\top\), whereas the true Fisher information is \(F = \frac{1}{n}\sum_i \lambda_i\, \nabla\eta_i \nabla\eta_i^\top\). The running average of \(g \odot g\) in Adam is, per observation, the diagonal of \(\hat F\), so Adam acts as a diagonal preconditioner built from \(\hat F^{1/2}\), a cheap stand-in for curvature. It is the square root, not the inverse, so it is not a natural-gradient step (which would use \(F^{-1}\)); and because the app feeds Adam the full-data gradient, \(\hat v_k\) here averages \(g_k^2\) over iterations rather than over observations, so the reading as \(\hat F\) is an approximation. The line-search methods keep the gradient but choose the step by the Wolfe conditions, so there is no learning rate to tune. Conjugate gradient builds its direction from the previous one, which on a quadratic makes successive directions conjugate with respect to the Hessian. BFGS learns an approximation of the inverse Hessian from the gradient differences alone, the only method here with a full-matrix preconditioner; compare Newton’s method, \(P_k = H^{-1}\), and Fisher scoring, \(P_k = F^{-1}\).

Epochs and mini-batches. For SGD one iteration is one epoch: a shuffled pass of \(\lceil n/m \rceil\) mini-batch steps over all the data. An SGD iteration therefore costs about as many gradient evaluations as one full-batch step of the other methods, and the iteration axes are directly comparable. With the batch size equal to \(n\) SGD reduces to the fixed-rate method; the smaller the batch, the more updates each epoch makes, at the price of noisier steps. The loss and gradient norm shown are always those of the full data. The sample size goes up to 2000: the larger \(n\) is, the more mini-batch updates an epoch holds (\(n/m\)), and the clearer SGD’s advantage over the full-batch methods becomes. To keep the browser responsive an SGD run stops after 20000 mini-batch steps in all, which the info line reports. For these thirteen parameters BFGS is hard to beat; momentum, Adam and SGD are what larger networks use, where a Hessian approximation is too expensive.

The gradient. The lower network has the same layout, with each edge and intercept box coloured by the corresponding component of the gradient of the loss, computed by backpropagation. Its colour scale is set by the largest component at the current iteration, so you can see which weights the method is still pushing on as the run converges.

The fit. The right panel shows the generated counts, the true mean curve (dashed red, with its two modes), the mean curve at the starting weights (grey) and the current fit (green). Because \(\log\lambda = v_0 + \sum_j v_j h_j\), the units combine by adding in the exponent, not by adding curves, so the second plot shows the fit on the log scale: the four scaled units \(v_j h_j\) (dotted; orange, blue, black and cyan for units 1 to 4) simply add up to \(\log\lambda\) (green), and the true \(\log\lambda\) (dashed red) is the target. The box above the plots hides this second plot. Beneath the fit are the loss and the gradient norm per iteration. The top row of controls sets up the data (sample size and a new dataset), the optimizer, the maximum number of iterations and the colour contrast; the second row holds the step rule of the chosen method, the playback buttons, and a slider that shows the fit at any iteration of the run (it follows the playback, and you can drag it to scrub through the run). The loss is not convex, so different starts can end at different fits; try Random a few times.

## units 3, 4 over (0.8, 2.2), so lambda(x) has two modes
TRUE_THETA <- c(2.5, 2.5, 2.5, 2.5,   5.5, 2, -2, -5.5,   1.2, -1.2, 1.0, -1.0,   -0.5)

fwd <- function(th, x) {
  H <- tanh(outer(x, th[1:J]) + rep(th[J + 1:J], each = length(x)))
  list(H = H, eta = as.vector(th[3 * J + 1] + H %*% th[2 * J + 1:J]))
}
lambda_fun <- function(th, x) exp(fwd(th, x)$eta)

## the scaled units v_j h_j(x), one column per hidden unit
unit_terms <- function(th, x) sweep(fwd(th, x)$H, 2, th[2 * J + 1:J], "*")

## average Poisson negative log-likelihood (the log y! term dropped)
nll <- function(th, x, y) {
  e <- fwd(th, x)$eta
  v <- mean(exp(e) - y * e)
  if (is.finite(v)) v else Inf
}

## its gradient by backpropagation:  delta_eta -> delta_a -> weights
nll_grad <- function(th, x, y) {
  E <- fwd(th, x)
  d_eta <- exp(E$eta) - y                                   # n
  d_a   <- (1 - E$H^2) * outer(d_eta, th[2 * J + 1:J])      # n x J
  c(colSums(d_a * x),                                       # w_j1
    colSums(d_a),                                           # w_j0
    as.vector(crossprod(E$H, d_eta)),                       # v_j
    sum(d_eta)) / length(x)                                 # v_0
}

## ------------------------------------------------------- Wolfe line search
line_search <- function(f, g, x, d, a0 = 1, c1 = 1e-4, c2 = 0.9) {
  f0 <- f(x); gd0 <- sum(g(x) * d)
  if (!is.finite(gd0) || gd0 >= 0) return(list(alpha = 0, ok = FALSE))
  lo <- 0; hi <- Inf; a <- a0
  for (it in 1:60) {
    fa <- f(x + a * d)
    if (!is.finite(fa) || fa > f0 + c1 * a * gd0) {
      hi <- a; a <- 0.5 * (lo + hi)
    } else {
      if (sum(g(x + a * d) * d) < c2 * gd0) {
        lo <- a
        a <- if (is.finite(hi)) 0.5 * (lo + hi) else min(2 * a, 1e4)
      } else {
        return(list(alpha = a, ok = TRUE))
      }
    }
    if (a < 1e-14) return(list(alpha = 0, ok = FALSE))
  }
  list(alpha = a, ok = (a > 0))
}

## ------------------------------------------------------------- algorithms
## Row k of P, G, F is the state after k - 1 steps; A[k] is the step length
## that led to it.
run_alg <- function(alg, th0, x, y, maxit, lr = 0.1, mom = 0.9, m = 20,
                    tol = 1e-6) {
  p <- length(th0)
  f <- function(th) nll(th, x, y)
  g <- function(th) nll_grad(th, x, y)
  P <- G <- matrix(NA_real_, maxit + 1, p)
  Fv <- AL <- rep(NA_real_, maxit + 1)
  th <- th0; B <- diag(p)
  g_prev <- d_prev <- s_prev <- NULL
  n <- 0; note <- "maximum number of iterations reached"
  ## state for the fixed-rate rules
  n_obs <- length(x); m <- max(1, min(m, n_obs))
  capped <- FALSE
  if (alg == "sgd") {                        # keep the number of mini-batch steps in check
    cap <- max(1, floor(20000 / ceiling(n_obs / m)))
    if (maxit > cap) { maxit <- cap; capped <- TRUE }
  }
  vel <- m1 <- m2 <- numeric(p)              # momentum velocity; Adam moments
  if (alg == "sgd") set.seed(11)

  for (k in 1:(maxit + 1)) {
    gk <- g(th); fk <- f(th)
    P[k, ] <- th; G[k, ] <- gk; Fv[k] <- fk; n <- k
    if (!is.finite(fk) || !all(is.finite(gk))) { note <- "diverged"; break }
    if (sqrt(sum(gk^2)) < tol) { note <- "converged: gradient norm < 1e-6"; break }
    if (k > maxit) break

    if (alg %in% c("gdfix", "mom", "adam", "sgd")) {  # fixed rate, no line search
      if (alg == "sgd") {                    # one iteration is one epoch: a shuffled
        perm <- sample.int(n_obs)            # pass of mini-batch steps over all the
        for (s0 in seq(1, n_obs, by = m)) {  # data, so it is paired with a full-batch step
          idx <- perm[s0:min(s0 + m - 1, n_obs)]
          th <- th - lr * nll_grad(th, x[idx], y[idx])
        }
        AL[k + 1] <- lr
        next
      } else gs <- gk
      if (alg == "mom") {                    # heavy ball: keep a fraction of the last step
        vel <- mom * vel - lr * gs
        dth <- vel
      } else if (alg == "adam") {            # per-weight step from running moments
        m1 <- 0.9 * m1 + 0.1 * gs
        m2 <- 0.999 * m2 + 0.001 * gs^2
        dth <- -lr * (m1 / (1 - 0.9^k)) / (sqrt(m2 / (1 - 0.999^k)) + 1e-8)
      } else dth <- -lr * gs                 # gradient descent / SGD
      AL[k + 1] <- lr
      th <- th + dth
      next
    }

    if (alg == "gd") {
      d <- -gk
    } else if (alg == "cg") {
      if (is.null(d_prev)) {
        d <- -gk
      } else {
        yk <- gk - g_prev
        beta <- max(0, sum(gk * yk) / max(sum(g_prev^2), 1e-300))   # PR+
        d <- -gk + beta * d_prev
      }
      if (sum(gk * d) > -1e-12) d <- -gk                           # restart
    } else {                                                       # BFGS
      if (!is.null(s_prev)) {
        yk <- gk - g_prev; sy <- sum(s_prev * yk)
        if (sy > 1e-10) {
          if (k == 2) B <- (sy / sum(yk^2)) * diag(p)    # scale the first B
          rho <- 1 / sy
          V <- diag(p) - rho * outer(s_prev, yk)
          B <- V %*% B %*% t(V) + rho * outer(s_prev, s_prev)
        }
      }
      d <- -as.vector(B %*% gk)
      if (sum(gk * d) > -1e-12) { B <- diag(p); d <- -gk }
    }

    a0 <- if (alg == "bfgs" && k > 1) 1 else min(1, 1 / sqrt(sum(d^2)))
    ls <- line_search(f, g, th, d, a0)
    if (!ls$ok) { note <- "line search failed"; break }
    AL[k + 1] <- ls$alpha
    s_prev <- ls$alpha * d; g_prev <- gk; d_prev <- d
    th <- th + s_prev
  }
  if (capped && grepl("^maximum", note))
    note <- paste0("stopped after ", maxit, " epochs (20000 mini-batch steps)")
  i <- seq_len(n)
  list(P = P[i, , drop = FALSE], G = G[i, , drop = FALSE], F = Fv[i],
       A = AL[i], n = n, note = note)
}

  title(main = main, cex.main = 1.05)
  H <- cbind(XH, YH)
  ## edges, each carrying one weight: x -> h_j (w_j1), h_j -> output (v_j)
  for (j in 1:J) {
    segments(NX[1], NX[2], H[j, 1], H[j, 2], col = col[j],     lwd = lw[j],     lend = 1)
    segments(H[j, 1], H[j, 2], NO[1], NO[2], col = col[2 * J + j], lwd = lw[2 * J + j], lend = 1)
  }
  ## intercept nodes: a thin arrow from each box to the neuron it shifts
  tgt <- rbind(H, NO)
  for (i in seq_along(BIASIDX)) {
    b <- ADJ[BIASIDX[i], ]
    dy <- sign(tgt[i, 2] - b$y)
    if (i == J + 1) arrows(b$x, b$y + 0.03, NO[1] - 0.03, NO[2] - 0.03, length = 0.08,
                       col = "grey45", lwd = 1.4)
    else arrows(b$x, b$y + dy * 0.035, tgt[i, 1], tgt[i, 2] - dy * 0.035,
                length = 0.08, col = "grey45", lwd = 1.4)
  }
  ## nodes
  points(rbind(NX, H, NO), pch = 21, cex = 3.2, bg = "white", lwd = 1.6)
  text(NX[1], NX[2], "x", font = 2)
  text(H[, 1], H[, 2], paste0("h", 1:J), cex = 0.8)
  text(NO[1], NO[2], expression(lambda), cex = 1.1)
  ## values: an intercept is a box coloured by its value; an edge weight is
  ## plain text on a white patch so the edge does not strike through it
  for (i in 1:NP) {
    b <- ADJ[i, ]; isb <- i %in% BIASIDX
    if (isb) rect(b$x - 0.075, b$y - 0.036, b$x + 0.075, b$y + 0.036,
                  col = col[i], border = "grey35")
    else rect(b$x - 0.072, b$y - 0.028, b$x + 0.072, b$y + 0.028,
              col = adjustcolor("white", 0.8), border = NA)
    dark <- abs(tone(v[i], vmax, k)) > 0.62
    text(b$x, b$y, sprintf(paste0("%s=%.", digits, "f"), b$sym, v[i]),
         cex = 0.85, font = if (isb) 1 else 2,
         col = if (isb) { if (dark) "white" else "black" }
               else if (dark) col[i] else "grey20")
  }
  if (arrows) for (r in seq_len(nrow(ARROWS))) {
    a <- ARROWS[r, ]
    if (a$horiz)                                   # left / right: -, +
      polygon(a$ax + a$dir * c(-0.013, -0.013, 0.015),
              a$ay + c(-0.033, 0.033, 0), col = "grey30", border = NA)
    else                                           # down / up: -, +
      polygon(a$ax + c(-0.026, 0.026, 0), a$ay + a$dir * c(-0.016, -0.016, 0.02),
              col = "grey30", border = NA)
  }
  ## signed colour bar
  xs <- seq(0.12, 1.08, length.out = 201)
  rect(xs, -0.24, xs + diff(xs)[1], -0.19, col = DIV, border = NA)
  lab <- formatC(vmax, format = "g", digits = 2)
  text(c(0.12, 0.60, 1.08), -0.16, c(paste0("-", lab), "0", paste0("+", lab)),
       cex = 0.72)
}

## ----------------------------------------------------------------- server
server <- function(input, output, session) {

  dseed <- reactiveVal(1); rseed <- reactiveVal(1)
  theta0 <- reactiveVal(rand_start(1))        # the starting weights
  playing <- reactiveVal(FALSE)
  kpos <- reactiveVal(0L)                     # current iteration, 0 .. R$n - 1

  observeEvent(input$newdata, dseed(dseed() + 1))
  ## a sensible starting learning rate for each fixed-rate method
  observeEvent(input$alg, {
    lr0 <- c(gdfix = 0.1, sgd = 0.1, mom = 0.05, adam = 0.1)[input$alg]
    if (!is.na(lr0)) updateSliderInput(session, "lr", value = unname(lr0))
  })
  observeEvent(input$settrue, theta0(TRUE_THETA))
  observeEvent(input$setzero, theta0(rep(0, NP)))
  observeEvent(input$newstart, {
    rseed(rseed() + 1); theta0(rand_start(rseed()))
  })
  ## an arrow click moves the nearest arrow's weight by one step
  observeEvent(input$netv_click, {
    cl <- input$netv_click
    d <- ((cl$x - ARROWS$ax) / 0.05)^2 + ((cl$y - ARROWS$ay) / 0.04)^2
    i <- which.min(d)
    if (d[i] < 2) {
      th <- theta0()
      th[ARROWS$idx[i]] <- th[ARROWS$idx[i]] + ARROWS$dir[i] * as.numeric(input$astep)
      theta0(th)
    }
  })

  DAT <- reactive(make_data(input$n, dseed()))
  run <- reactive({
    D <- DAT()
    fixed <- input$alg %in% c("gdfix", "sgd", "mom", "adam")
    run_alg(input$alg, theta0(), D$x, D$y, as.numeric(input$maxit),
            lr  = if (fixed) input$lr else NA,
            mom = if (input$alg == "mom") input$mom else NA,
            m   = if (input$alg == "sgd") as.numeric(input$batch) else NA)
  })

  observeEvent(run(), {
    kpos(0L)
    updateSliderInput(session, "kshow", min = 0, max = run()$n - 1L, value = 0)
    playing(isTRUE(input$autoplay) && run()$n > 1)
  })

  ## the iteration slider follows the run, and scrubs it when dragged
  observeEvent(kpos(), updateSliderInput(session, "kshow", value = kpos()))
  observeEvent(input$kshow, {
    if (!isTRUE(playing()) && input$kshow != kpos())
      kpos(min(as.integer(input$kshow), run()$n - 1L))
  })

  observeEvent(input$play, {
    if (isTRUE(playing())) playing(FALSE)
    else {
      if (kpos() >= run()$n - 1) kpos(0L)
      playing(TRUE)
    }
  })
  observeEvent(playing(), {
    updateActionButton(session, "play",
                       icon = icon(if (isTRUE(playing())) "pause" else "play"))
  })
  observe({
    if (!isTRUE(playing())) return()
    kmax <- isolate(run()$n) - 1
    k <- isolate(kpos())
    if (k >= kmax) { playing(FALSE); return() }
    invalidateLater(300, session)
    kpos(k + 1L)
  })
  observeEvent(input$step,  { playing(FALSE); kpos(min(kpos() + 1L, run()$n - 1L)) })
  observeEvent(input$back,  { playing(FALSE); kpos(max(kpos() - 1L, 0L)) })
  observeEvent(input$first, { playing(FALSE); kpos(0L) })
  observeEvent(input$toend, { playing(FALSE); kpos(run()$n - 1L) })

  cur <- reactive({
    R <- run(); k <- min(kpos(), R$n - 1L) + 1L
    list(R = R, k = k, th = R$P[k, ], g = R$G[k, ])
  })

  output$netv <- renderPlot({
    C <- cur(); R <- C$R
    vmax <- max(3, quantile(abs(R$P), 0.95))
    par(mar = c(1, 1, 3, 1))
    draw_net(C$th, vmax, as.numeric(input$ctr),
             sprintf("weights, iteration %d   (loss %.4f)", C$k - 1L, R$F[C$k]))
  })

  output$netg <- renderPlot({
    C <- cur()
    gmax <- max(1e-8, max(abs(C$g)))
    par(mar = c(1, 1, 3, 1))
    draw_net(C$g, gmax, as.numeric(input$ctr),
             sprintf("gradient of the loss, iteration %d", C$k - 1L),
             arrows = FALSE, digits = 3)
  })

  ## one clearly different colour per hidden unit
  PCOL <- c("#F28E00", "#1F5FBF", "black", "#00B5CC")

  output$fit <- renderPlot({
    C <- cur(); R <- C$R; D <- DAT()
    xg <- seq(-3, 3, length.out = 300)
    par(mar = c(4, 4.2, 2.5, 1))
    plot(D$x, D$yj, pch = 19, cex = 0.7,
         col = adjustcolor("#E8820C", 0.5), ylim = c(0, max(D$y) + 1),
         xlab = "x", ylab = expression(lambda(x) ~ "and counts y"),
         main = "fitted mean and the generated data")
    lines(xg, lambda_fun(TRUE_THETA, xg), col = "red", lty = 2, lwd = 2)
    lines(xg, lambda_fun(R$P[1, ], xg), col = "grey60", lwd = 2)
    lines(xg, lambda_fun(C$th, xg), col = "#2E8B57", lwd = 3)
    leg <- data.frame(
      lab = c("true mean", "starting weights", "current fit", "counts"),
      lwd = c(2, 2, 3, NA), lty = c(2, 1, 1, NA), pch = c(NA, NA, NA, 19),
      col = c("red", "grey60", "#2E8B57", adjustcolor("#E8820C", 0.7)),
      stringsAsFactors = FALSE)
    legend("topright", bty = "n", cex = 0.78, lwd = leg$lwd, lty = leg$lty,
           pch = leg$pch, col = leg$col, legend = leg$lab)
  })

  ## the same fit on the log scale, where the units simply add up
  output$eta <- renderPlot({
    C <- cur(); xg <- seq(-3, 3, length.out = 300)
    VH <- unit_terms(C$th, xg)
    eta <- C$th[3 * J + 1] + rowSums(VH)
    eta_true <- log(lambda_fun(TRUE_THETA, xg))
    par(mar = c(4, 4.2, 2.5, 1))
    matplot(xg, VH, type = "l", lty = 3, lwd = 2.5, col = PCOL,
            ylim = pmax(-8, pmin(8, range(VH, eta, eta_true))),
            xlab = "x", ylab = expression(log ~ lambda(x)),
            main = "log scale: the scaled units v_j h_j add up to the fit")
    abline(h = 0, lty = 3, col = "grey70")
    lines(xg, eta_true, col = "red", lty = 2, lwd = 2)
    lines(xg, eta, col = "#2E8B57", lwd = 3)
    legend("topright", bty = "n", cex = 0.78, ncol = 2,
           lwd = c(rep(2.5, J), 3, 2), lty = c(rep(3, J), 1, 2),
           col = c(PCOL, "#2E8B57", "red"),
           legend = c(paste0("v", 1:J, " h", 1:J), "sum + v0 (fit)", "true"))
  })

  output$conv <- renderPlot({
    C <- cur(); R <- C$R
    it <- 0:(R$n - 1)
    gn <- sqrt(rowSums(R$G^2))
    par(mfrow = c(1, 2), mar = c(4, 4.2, 2.5, 1))
    plot(it, R$F, type = "b", pch = 20, xlab = "iteration", ylab = "loss",
         main = "average negative log-likelihood")
    points(it[C$k], R$F[C$k], pch = 19, col = "red", cex = 1.6)
    plot(it, log10(gn), type = "b", pch = 20, xlab = "iteration",
         ylab = expression(log[10] ~ "||gradient||"), main = "gradient norm")
    points(it[C$k], log10(gn[C$k]), pch = 19, col = "red", cex = 1.6)
  })

  output$info <- renderUI({
    C <- cur(); R <- C$R
    a <- R$A[C$k]
    HTML(paste0(
      "<b>", if (input$alg == "sgd") "epoch" else "iteration", "</b> ", C$k - 1L, " of ", R$n - 1L,
      " &nbsp; <b>loss</b> ", formatC(R$F[C$k], digits = 5, format = "f"),
      " &nbsp; <b>||gradient||</b> ", formatC(sqrt(sum(C$g^2)), digits = 3, format = "g"),
      if (is.finite(a)) paste0(" &nbsp; <b>",
                               if (input$alg %in% c("gdfix", "sgd", "mom", "adam"))
                                 "learning rate" else "step length", "</b> ",
                               formatC(a, digits = 3, format = "g")) else "",
      "<br><i>", R$note, "</i>"))
  })
}
